const App = () => {
  const [isSidebarOpen, setIsSidebarOpen] = useState(true);
  const [sidebarSide, setSidebarSide] = useState('left');
  const [isDarkMode, setIsDarkMode] = useState(false);

  const [panelItem, setPanelItem] = useState(null); 
  const [isSearchExpanded, setIsSearchExpanded] = useState(false);

  const [useOverlaySearch, setUseOverlaySearch] = useState(false);

    "Admin": { primary: "emerald-500", bg: "bg-emerald-500", text: "text-emerald-500", border: "border-emerald-500", hoverBg: "hover:bg-emerald-50", ring: "ring-emerald-500" },
    "Analytics": { primary: "orange-500", bg: "bg-orange-500", text: "text-orange-500", border: "border-orange-500", hoverBg: "hover:bg-orange-50", ring: "ring-orange-500" }
  };

  const theme = appThemes[currentAppName] || appThemes["ClassBank"];
  const popoutRef = useRef(null);
  const searchRef = useRef(null);
  const isLeft = sidebarSide === 'left';

  // --- Core & Banking State ---
  const [viewMode, setViewMode] = useState('grid');
  const [bankViewMode, setBankViewMode] = useState('grid'); 
  const [selectedStudent, setSelectedStudent] = useState(null); 
  const [editedPin, setEditedPin] = useState(""); 
  const [isEditingName, setIsEditingName] = useState(false); 
  const [editedName, setEditedName] = useState("");
  const [selectedQrStudent, setSelectedQrStudent] = useState(null); 
  const [transactionModal, setTransactionModal] = useState(null); 
  const [classDeposit, setClassDeposit] = useState(null); 
  const [paydayModal, setPaydayModal] = useState(null); 
  const [studentToDelete, setStudentToDelete] = useState(null); // State for Delete Confirmation
  
  // --- Jobs State ---
  const [jobsList, setJobsList] = useState(INITIAL_JOBS);
  const [manageJobModal, setManageJobModal] = useState(null); 
  const [assignJobModal, setAssignJobModal] = useState(null); 
  const [openTxMenuId, setOpenTxMenuId] = useState(null); // State for transaction row menus

  // --- Behavior State ---
  const [behaviorModal, setBehaviorModal] = useState(null); 
  const [syncBehaviorToBank, setSyncBehaviorToBank] = useState(false); 

  // --- Pending Approvals State ---
  const [pendingApprovals, setPendingApprovals] = useState([
    { id: 'pa1', studentId: 4, studentName: 'Dakota Smith', studentAvatar: '🎨', date: '10 mins ago', description: 'Class Store: Fancy Eraser', amount: 15, type: 'deduct' },
    { id: 'pa2', studentId: 12, studentName: 'Lily Chen', studentAvatar: '🌸', date: '1 hour ago', description: 'Found a lost book', amount: 5, type: 'deposit' }
  ]);

  const [studentsData, setStudentsData] = useState(() => {
    const goals = [{ item: "Homework Pass", amount: 150 }, { item: "Pizza Slice", amount: 300 }, { item: "Extra Recess", amount: 200 }, { item: "Prize Box", amount: 100 }];
    
    return STUDENT_DATABASE.map((student, index) => {
      const assignedJob = INITIAL_JOBS[Math.floor(Math.random() * INITIAL_JOBS.length)];
      return {

        isPinVisible: false,
        balance: Math.floor(Math.random() * 300) + 20,
        behaviorPoints: Math.floor(Math.random() * 10),
        job: assignedJob.title,
        salary: assignedJob.salary,
        goal: goals[Math.floor(Math.random() * goals.length)],
        transactions: [
          { id: Date.now() - (index * 1000) - 10000, date: 'Today', description: 'Class Store Purchase', amount: 15, type: 'deduct' },
          { id: Date.now() - (index * 1000) - 86400000, date: 'Yesterday', description: 'Weekly Allowance', amount: 50, type: 'deposit' },
          { id: Date.now() - (index * 1000) - 172800000, date: 'Oct 24', description: 'Homework Bonus', amount: 10, type: 'deposit' }
        ]
      };
    });
  });

  const togglePinVisibility = (id) => {
    setStudentsData(prev => prev.map(s => 
      s.id === id ? { ...s, isPinVisible: !s.isPinVisible } : s
    ));
  };

  const savePin = () => {
    if (editedPin.length === 4 && selectedStudent) {
      setStudentsData(prev => prev.map(s => 
        s.id === selectedStudent.id ? { ...s, pin: editedPin } : s
      ));
      setSelectedStudent(prev => ({ ...prev, pin: editedPin }));
    }
  };

  const saveName = () => {
    if (editedName.trim() && selectedStudent) {
      setStudentsData(prev => prev.map(s => 
        s.id === selectedStudent.id ? { ...s, name: editedName.trim() } : s
      ));
      setSelectedStudent(prev => ({ ...prev, name: editedName.trim() }));
      setIsEditingName(false);
    }
  };

  const executeTransaction = () => {
    const amount = parseFloat(transactionModal.amount);
    if (isNaN(amount) || amount <= 0) {
      setTransactionModal(prev => ({ ...prev, error: 'Please enter a valid amount.' }));
      return;
    }
    
    if (transactionModal.mode === 'pin' && transactionModal.pin !== transactionModal.student.pin) {
      setTransactionModal(prev => ({ ...prev, error: 'Incorrect PIN. Try again.' }));
      return;
    }

    const newTx = {
      id: Date.now(),
      date: 'Just now',
      description: transactionModal.description || (transactionModal.type === 'deposit' ? 'Deposit' : 'Deduction'),
      amount: amount,
      type: transactionModal.type
    };

    setStudentsData(prev => prev.map(s => {
      if (s.id === transactionModal.student.id) {
        return {
          ...s,
          balance: transactionModal.type === 'deposit' ? s.balance + amount : s.balance - amount,
          transactions: [newTx, ...(s.transactions || [])]
        };
      }
      return s;
    }));
    
    setTransactionModal(null);
  };

  const executeClassDeposit = () => {
    const amount = parseFloat(classDeposit.amount);
    if (isNaN(amount) || amount <= 0) {
      setClassDeposit(prev => ({ ...prev, error: 'Please enter a valid amount.' }));
      return;
    }
    
    if (classDeposit.teacherPin !== '1234') {
      setClassDeposit(prev => ({ ...prev, error: 'Incorrect Teacher PIN. (Hint: use 1234)' }));
      return;
    }

    const newTx = {
      id: Date.now(),
      date: 'Just now',
      description: classDeposit.description || 'Class Deposit',
      amount: amount,
      type: 'deposit'
    };

    setStudentsData(prev => prev.map(s => ({
      ...s,
      balance: s.balance + amount,
      transactions: [newTx, ...(s.transactions || [])]
    })));
    
    setClassDeposit(null);
  };

  const executePayday = () => {
    if (paydayModal.teacherPin !== '1234') {
      setPaydayModal(prev => ({ ...prev, error: 'Incorrect Teacher PIN. (Hint: use 1234)' }));
      return;
    }
    
    if (paydayModal.selectedStudents.length === 0) {
      setPaydayModal(prev => ({ ...prev, error: 'Please select at least one student.' }));
      return;
    }

    setStudentsData(prev => prev.map(s => {
      if (!paydayModal.selectedStudents.includes(s.id)) return s; 
      
      const salaryAmount = s.salary || 0;
      if (salaryAmount === 0) return s; 
      
      const newTx = {
        id: Date.now() + s.id, 
        date: 'Just now',
        description: `Payday: ${s.job}`,
        amount: salaryAmount,
        type: 'deposit'
      };
      
      return {
        ...s,
        balance: s.balance + salaryAmount,
        transactions: [newTx, ...(s.transactions || [])]
      };
    }));
    
    setPaydayModal(null);
  };

  const executeBehavior = (behavior) => {
    if (!behaviorModal || behaviorModal.selectedStudents.length === 0) return;

    setStudentsData(prev => prev.map(s => {
      if (behaviorModal.selectedStudents.includes(s.id)) {
         let updatedStudent = {
           ...s,
           behaviorPoints: (s.behaviorPoints || 0) + behavior.points
         };

         if (syncBehaviorToBank) {
            const amount = Math.abs(behavior.points); 
            const isDeposit = behavior.points > 0;
            const newTx = {
              id: Date.now() + s.id + Math.random(), 
              date: 'Just now',
              description: `Behavior: ${behavior.name}`,
              amount: amount,
              type: isDeposit ? 'deposit' : 'deduct'
            };
            updatedStudent.balance = isDeposit ? s.balance + amount : Math.max(0, s.balance - amount); 
            updatedStudent.transactions = [newTx, ...(s.transactions || [])];
         }
         return updatedStudent;
      }
      return s;
    }));
    
    setBehaviorModal(null);
  };

  const executeDeleteStudent = () => {
    if (!studentToDelete) return;
    
    // Remove student from database
    setStudentsData(prev => prev.filter(s => s.id !== studentToDelete.id));
    
    // Close modal and return to roster
    setStudentToDelete(null);
    setSelectedStudent(null);
  };

  const executeManageJob = () => {
    if (!manageJobModal.job.title || manageJobModal.job.salary === '') {
       setManageJobModal(prev => ({ ...prev, error: 'Title and Base Salary are required.' }));
       return;
    }
    
    if (manageJobModal.mode === 'create') {
      setJobsList([...jobsList, manageJobModal.job]);
    } else {
      setJobsList(jobsList.map(j => j.id === manageJobModal.job.id ? manageJobModal.job : j));
      // Update students who had the old job title
      setStudentsData(prev => prev.map(s => 
        s.job === manageJobModal.oldTitle 
          ? { ...s, job: manageJobModal.job.title, salary: parseFloat(manageJobModal.job.salary) } 
          : s
      ));
    }
    setManageJobModal(null);
  };

  const executeDeleteJob = () => {
    setJobsList(jobsList.filter(j => j.id !== manageJobModal.job.id));
    // Set students with this job to unassigned
    setStudentsData(prev => prev.map(s => 
      s.job === manageJobModal.oldTitle 
        ? { ...s, job: 'Unassigned', salary: 0 } 
        : s
    ));
    setManageJobModal(null);
  };

  const executeAssignJob = () => {
    setStudentsData(prev => prev.map(s => {
      const isSelected = assignJobModal.selectedStudents.includes(s.id);
      if (isSelected && s.job !== assignJobModal.job.title) {
        // Give them this job
        return { ...s, job: assignJobModal.job.title, salary: parseFloat(assignJobModal.job.salary) };
      } else if (!isSelected && s.job === assignJobModal.job.title) {
        // Remove this job if they had it
        return { ...s, job: 'Unassigned', salary: 0 };
      }
      return s;
    }));
    setAssignJobModal(null);
  };

  const executeUndoTransaction = (studentId, transactionId) => {
    setStudentsData(prev => prev.map(s => {
      if (s.id !== studentId) return s;
      
      const txToUndo = s.transactions?.find(tx => tx.id === transactionId);
      if (!txToUndo) return s;
      
      // Reverse the operation
      const balanceChange = txToUndo.type === 'deposit' ? -txToUndo.amount : txToUndo.amount;
      
      return {
        ...s,
        balance: Math.max(0, s.balance + balanceChange),
        transactions: s.transactions.filter(tx => tx.id !== transactionId)
      };
    }));
    setOpenTxMenuId(null);
  };

  const handleApproveTransaction = (request) => {
    setStudentsData(prev => prev.map(s => {
      if (s.id === request.studentId) {
        const newTx = {
          id: Date.now(),
          date: 'Just now',
          description: request.description,
          amount: request.amount,
          type: request.type
        };
        return {
          ...s,
          balance: request.type === 'deposit' ? s.balance + request.amount : Math.max(0, s.balance - request.amount),
          transactions: [newTx, ...(s.transactions || [])]
        };
      }
      return s;
    }));
    // Remove from pending
    setPendingApprovals(prev => prev.filter(req => req.id !== request.id));
  };

  const handleDenyTransaction = (requestId) => {
    setPendingApprovals(prev => prev.filter(req => req.id !== requestId));
  };
  // -----------------------------

  useEffect(() => {
    const handleResize = () => {
      setUseOverlaySearch(window.innerWidth < 768);
    };
    handleResize();
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, []);

  useEffect(() => {
    const handleClickOutside = (event) => {
      if (popoutRef.current && !popoutRef.current.contains(event.target)) {
        setPopoutItem(null);
      }
      if (searchRef.current && !searchRef.current.contains(event.target) && !searchQuery) {
        setIsSearchExpanded(false);
      }
    };
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, [searchQuery]);

  const navigationItems = [
    { id: 'bank', name: 'Bank', icon: Landmark, type: 'link' },
    { id: 'behavior', name: 'Behavior', icon: Award, type: 'link' },
    { id: 'jobs', name: 'Jobs', icon: Briefcase, type: 'link' },
    { id: 'transactions', name: 'Transactions', icon: ArrowLeftRight, type: 'link' },
    { id: 'students', name: 'Students', icon: Users, type: 'link' }
  ];

  const sidebarWidth = isSidebarOpen ? 'w-64' : 'w-20';
  const navHeight = 'h-16';

  const handleItemClick = (item) => {
    if (item.type === 'link') {
      setActiveTab(item.name);
      setExpandedItem(null);
      setPanelItem(null);
    } else if (item.type === 'accordion') {
      if (!isSidebarOpen) setIsSidebarOpen(true);
      setExpandedItem(expandedItem === item.id ? null : item.id);
      setPanelItem(null);
    } else if (item.type === 'popout') {
      setPopoutItem(popoutItem === item.id ? null : item.id);
    } else if (item.type === 'panel') {
      const isOpening = panelItem !== item.id;
      setPanelItem(isOpening ? item.id : null);
      setExpandedItem(null);
      if (isOpening) setActiveTab(null);
    }
  };

  return (
    <div className={`flex h-screen overflow-hidden font-sans transition-all duration-300 ${isDarkMode ? 'bg-slate-950 text-slate-100' : 'bg-slate-50 text-slate-900'} ${isLeft ? 'flex-row' : 'flex-row-reverse'}`}>
      
      {/* 1. PRIMARY SIDEBAR */}
      <aside className={`relative z-50 transition-all duration-300 ease-in-out flex flex-col shadow-sm ${sidebarWidth} ${isLeft ? 'border-r' : 'border-l'} ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
        <div className={`flex items-center justify-center shrink-0 border-b ${navHeight} w-full ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
          <div className="flex items-center justify-center px-4">
            {isSidebarOpen ? <LogoHorizontal className="h-6 w-auto" /> : <LogoIcon2x2 className="w-8 h-8" />}
          </div>
        </div>

        <div className="pt-2 shrink-0"></div>

        <nav className="flex-1 px-2 space-y-1 pt-2 pb-4 relative overflow-y-auto overflow-x-hidden">
          {navigationItems.map((item) => {
            const isTabActive = activeTab === item.name || (item.type === 'accordion' && item.subItems.includes(activeTab));
            const isPanelOpen = panelItem === item.id;
            const isVisuallyActive = isTabActive || isPanelOpen;
            const isExpanded = expandedItem === item.id;
            const hasPopout = popoutItem === item.id;

            return (
              <div key={item.id} className="relative">
                <button
                  onClick={() => handleItemClick(item)}
                  className={`flex items-center transition-all group relative rounded-xl h-12
                    ${isSidebarOpen ? 'w-full px-3' : 'w-16 mx-auto justify-center'}
                    ${isVisuallyActive ? `${theme.bg} text-white shadow-lg` : `${isDarkMode ? 'text-slate-400 hover:bg-slate-800 hover:text-slate-200' : 'text-slate-500 hover:bg-slate-100 hover:text-slate-900'}`}`}
                >
                  <div className={`flex items-center justify-center shrink-0`}>
                    <item.icon size={20} strokeWidth={isVisuallyActive ? 2.5 : 2} />
                  </div>
                  {isSidebarOpen && (
                    <>
                      <span className="ml-4 font-semibold text-sm flex-1 text-left truncate">{item.name}</span>
                      {item.type === 'accordion' && (
                        <ChevronDown size={14} className={`transition-transform duration-200 shrink-0 ${isExpanded ? 'rotate-180' : ''}`} />
                      )}
                      {item.type === 'panel' && (
                        <ChevronRight size={14} className={`transition-all duration-200 shrink-0 ${isLeft ? '' : 'rotate-180'} ${isPanelOpen ? 'opacity-100' : 'opacity-0 group-hover:opacity-40'}`} />
                      )}
                      {item.type === 'popout' && <MoreVertical size={14} className="opacity-40 shrink-0" />}
                    </>
                  )}
                  {!isSidebarOpen && (
                    <div className={`absolute ${isLeft ? 'left-full ml-4' : 'right-full mr-4'} px-3 py-1.5 bg-slate-900 text-white text-xs font-bold rounded-md opacity-0 group-hover:opacity-100 pointer-events-none transition-opacity z-[60] whitespace-nowrap shadow-xl`}>
                      {item.name}
                    </div>
                  )}
                </button>
                {item.type === 'accordion' && isExpanded && isSidebarOpen && (
                  <div className={`mt-1 space-y-1 ${isDarkMode ? 'border-slate-800' : 'border-slate-100'} ${isLeft ? 'ml-10 border-l-2' : 'mr-10 border-r-2'}`}>
                    {item.subItems.map(sub => (
                      <button 
                        key={sub}
                        onClick={() => setActiveTab(sub)}
                        className={`block w-full text-left py-2 px-4 text-xs font-medium transition-colors
                          ${isLeft ? 'rounded-r-lg' : 'rounded-l-lg text-right'}
                          ${activeTab === sub ? `${theme.text} ${isDarkMode ? 'bg-slate-800' : 'bg-slate-50'} font-bold` : `${isDarkMode ? 'text-slate-500 hover:text-slate-300 hover:bg-slate-800/50' : 'text-slate-400 hover:text-slate-600 hover:bg-slate-50/50'}`}`}

                      <div className="flex-1 min-w-0 pr-2">

'border-emerald-100 bg-emerald-50 hover:bg-emerald-100 dark:border-emerald-500/30 dark:bg-emerald-500/10 dark:hover:bg-emerald-500/20' : 'border-rose-100 bg-rose-50 hover:bg-rose-100 dark:border-rose-500/30 dark:bg-rose-500/10 dark:hover:bg-rose-500/20'}`}

              </h3>
              <p className={`text-sm mb-8 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>
                Are you sure you want to permanently remove <strong>{studentToDelete.name}</strong> from the roster? This action cannot be undone.
              </p>
              
              <div className="flex gap-3 w-full">
                <button 
                  onClick={() => setStudentToDelete(null)}
                  className={`flex-1 py-3 px-4 rounded-xl font-bold text-sm transition-colors ${isDarkMode ? 'bg-slate-800 text-slate-300 hover:bg-slate-700' : 'bg-slate-100 text-slate-600 hover:bg-slate-200'}`}
                >
                  Cancel
                </button>
                <button 
                  onClick={executeDeleteStudent}
                  className="flex-1 py-3 px-4 rounded-xl text-white font-bold text-sm bg-rose-500 hover:bg-rose-600 transition-colors shadow-sm"
                >
                  Yes, Remove
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* MANAGE JOB MODAL */}
      {manageJobModal && (
        <div 
          className="fixed inset-0 z-[140] flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm animate-in fade-in duration-200"
          onClick={() => setManageJobModal(null)}
        >
          <div 
            className={`relative w-full max-w-md p-6 md:p-8 rounded-3xl shadow-2xl animate-in zoom-in-95 duration-200 max-h-[90vh] overflow-y-auto ${isDarkMode ? 'bg-slate-900 border border-slate-700' : 'bg-white border border-slate-200'}`}
            onClick={(e) => e.stopPropagation()}
          >
            {/* Header */}
            <button 
              onClick={() => setManageJobModal(null)} 
              className={`absolute top-4 right-4 p-2 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:text-white hover:bg-slate-800' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}
            >
              <X size={20} />
            </button>

            <div className="flex items-center space-x-4 mb-6">
              <div className={`w-14 h-14 rounded-full flex items-center justify-center text-3xl shadow-inner shrink-0 text-white bg-indigo-500`}>
                <Briefcase size={28} />
              </div>
              <div className="min-w-0">
                <h3 className={`font-bold text-lg truncate ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>
                  {manageJobModal.mode === 'create' ? 'Create New Job' : 'Edit Job'}
                </h3>
                <p className={`text-sm truncate ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>
                  Configure role details and pay
                </p>
              </div>
            </div>

            {/* Title Field */}
            <div className="mb-4">
              <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Job Title</label>
              <div className={`flex items-center px-4 py-3 rounded-xl border transition-colors ${isDarkMode ? 'bg-slate-800 border-slate-700 focus-within:border-slate-500' : 'bg-slate-50 border-slate-200 focus-within:border-slate-400'}`}>
                <input 
                  type="text"
                  value={manageJobModal.job.title}
                  onChange={e => setManageJobModal({...manageJobModal, job: {...manageJobModal.job, title: e.target.value}, error: ''})}
                  placeholder="e.g., Door Holder"
                  className={`bg-transparent border-none outline-none w-full font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}
                  autoFocus
                />
              </div>
            </div>

            <div className="flex space-x-4 mb-6">
              {/* Icon Field */}
              <div className="flex-shrink-0 w-24 relative">
                <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Icon</label>
                <button 
                  onClick={() => setManageJobModal({...manageJobModal, showIconPicker: !manageJobModal.showIconPicker})}
                  className={`w-full flex items-center justify-center px-4 py-3 rounded-xl border transition-colors ${isDarkMode ? 'bg-slate-800 border-slate-700 hover:bg-slate-700' : 'bg-slate-50 border-slate-200 hover:bg-slate-100'}`}
                >
                  <span className="text-xl">{manageJobModal.job.icon}</span>
                  <ChevronDown size={14} className={`ml-2 opacity-50 transition-transform ${manageJobModal.showIconPicker ? 'rotate-180' : ''}`} />
                </button>

                {manageJobModal.showIconPicker && (
                  <div className={`absolute z-50 top-full mt-2 left-0 w-56 p-2 rounded-xl shadow-xl border grid grid-cols-5 gap-1 ${isDarkMode ? 'bg-slate-800 border-slate-700' : 'bg-white border-slate-200'}`}>
                    {['💼', '🚶', '📄', '💻', '📚', '🏦', '✉️', '🧽', '🧹', '🪴', '🗑️', '📝', '🚪', '💡', '🗣️', '🍎', '🎨', '🔋', '📫', '🧸', '🤝', '🎯', '👥', '🙋', '🧗', '💪', '🛠️', '🩺', '🔬', '🔧', '📦', '✏️', '🖍️', '📏', '📢'].map(emoji => (
                      <button
                        key={emoji}
                        onClick={() => setManageJobModal({...manageJobModal, job: {...manageJobModal.job, icon: emoji}, showIconPicker: false})}
                        className={`flex items-center justify-center p-1.5 text-xl rounded-lg transition-colors ${isDarkMode ? 'hover:bg-slate-700' : 'hover:bg-slate-100'}`}
                      >
                        {emoji}
                      </button>
                    ))}
                  </div>
                )}
              </div>

              {/* Salary Field */}
              <div className="flex-1">
                <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Weekly Salary</label>
                <div className={`flex items-center px-4 py-3 rounded-xl border transition-colors ${isDarkMode ? 'bg-slate-800 border-slate-700 focus-within:border-slate-500' : 'bg-slate-50 border-slate-200 focus-within:border-slate-400'}`}>
                  <DollarSign size={18} className="text-emerald-500 shrink-0" />
                  <input 
                    type="number"
                    value={manageJobModal.job.salary}
                    onChange={e => setManageJobModal({...manageJobModal, job: {...manageJobModal.job, salary: e.target.value}, error: ''})}
                    placeholder="0"
                    className={`bg-transparent border-none outline-none font-bold text-lg ml-2 w-full ${isDarkMode ? 'text-white' : 'text-slate-900'}`}
                  />
                </div>
              </div>
            </div>

            {manageJobModal.error && (
              <div className="mb-6 flex items-center text-xs font-bold text-rose-500">
                <AlertCircle size={14} className="mr-1.5 shrink-0" /> <span>{manageJobModal.error}</span>
              </div>
            )}

            {/* Actions */}
            <div className="flex gap-3">
              {manageJobModal.mode === 'edit' && (
                 <button 
                   onClick={executeDeleteJob}
                   className={`flex items-center justify-center p-4 rounded-xl border transition-colors ${isDarkMode ? 'border-slate-700 text-rose-500/80 hover:bg-slate-800' : 'border-slate-200 text-rose-500 hover:bg-rose-50'}`}
                   title="Delete Job"
                 >
                   <Trash2 size={20} />
                 </button>
              )}
              <button 
                onClick={executeManageJob}
                className="flex-1 flex items-center justify-center py-4 px-4 rounded-xl text-white font-bold transition-all shadow-sm bg-indigo-500 hover:bg-indigo-600"
              >
                Save Job
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ASSIGN JOB MODAL */}
      {assignJobModal && (
        <div 
          className="fixed inset-0 z-[140] flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm animate-in fade-in duration-200"
          onClick={() => setAssignJobModal(null)}
        >
          <div 
            className={`relative w-full max-w-md p-6 md:p-8 rounded-3xl shadow-2xl animate-in zoom-in-95 duration-200 max-h-[90vh] overflow-y-auto ${isDarkMode ? 'bg-slate-900 border border-slate-700' : 'bg-white border border-slate-200'}`}
            onClick={(e) => e.stopPropagation()}
          >
            {/* Header */}
            <button 
              onClick={() => setAssignJobModal(null)} 
              className={`absolute top-4 right-4 p-2 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:text-white hover:bg-slate-800' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}
            >
              <X size={20} />
            </button>

            <div className="text-center mb-6">
              <div className={`inline-flex w-16 h-16 rounded-full items-center justify-center text-3xl shadow-inner mb-4 ${isDarkMode ? 'bg-slate-800' : 'bg-slate-100'}`}>
                {assignJobModal.job.icon}
              </div>
              <h3 className={`font-bold text-2xl ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>
                Assign {assignJobModal.job.title}
              </h3>
            </div>

            {/* Student Selection List */}
            <div className="mb-6">
              <div className="flex items-center justify-between mb-2 px-1">
                <label className={`text-[10px] font-bold uppercase tracking-widest ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Assign to Students</label>
              </div>
              <div className={`max-h-60 overflow-y-auto rounded-xl border ${isDarkMode ? 'border-slate-700 bg-slate-800/30' : 'border-slate-200 bg-slate-50/50'}`}>
                {studentsData.map(student => (
                  <label key={`assign-${student.id}`} className={`flex items-center p-3 border-b last:border-0 cursor-pointer transition-colors ${isDarkMode ? 'border-slate-700 hover:bg-slate-800/50' : 'border-slate-200 hover:bg-slate-100/50'}`}>
                    <div className="flex items-center justify-center mr-3 pl-1">
                      <input 
                        type="checkbox" 
                        checked={assignJobModal.selectedStudents.includes(student.id)}
                        onChange={(e) => {
                          const isSelected = e.target.checked;
                          setAssignJobModal(prev => ({
                            ...prev,
                            selectedStudents: isSelected 
                              ? [...prev.selectedStudents, student.id]
                              : prev.selectedStudents.filter(id => id !== student.id)
                          }));
                        }}
                        className="w-4 h-4 rounded border-slate-300 text-indigo-500 focus:ring-indigo-500 bg-white dark:bg-slate-900 cursor-pointer"
                      />
                    </div>
                    <div className={`flex items-center justify-center w-8 h-8 rounded-full text-lg mr-3 shadow-sm ${isDarkMode ? 'bg-slate-900' : 'bg-white'}`}>
                      {student.avatar}
                    </div>
                    <div className="flex-1 min-w-0 pr-2">
                      <p className={`text-sm font-bold truncate ${isDarkMode ? 'text-slate-200' : 'text-slate-800'}`}>{student.name}</p>
                    </div>
                    {/* Show current job if they have a different one */}
                    {!assignJobModal.selectedStudents.includes(student.id) && student.job !== 'Unassigned' && (
                       <div className={`text-[10px] font-bold uppercase tracking-wider px-2 py-1 rounded-md ${isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-200 text-slate-500'}`}>
                          {student.job}
                       </div>
                    )}
                  </label>
                ))}
              </div>
            </div>

            {/* Submit Button */}
            <button 
              onClick={executeAssignJob}
              className="w-full flex items-center justify-center py-4 px-4 rounded-xl text-white font-bold transition-all shadow-sm bg-indigo-500 hover:bg-indigo-600"
            >
              Update Assignments
            </button>
          </div>
        </div>
      )}

    </div>
  );
};

export default App;
