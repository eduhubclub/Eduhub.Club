    <script>
        const apps = [
            { name: "Hub", color: "text-sky-500" },
            { name: "Class", color: "text-rose-500" },
            { name: "Bank", color: "text-emerald-500" },
            { name: "Group", color: "text-orange-500" },
            { name: "Randomizer", color: "text-purple-500" },
            { name: "Attendance", color: "text-teal-500" },
            { name: "Tie Breaker", color: "text-pink-500" },
            { name: "Calendar", color: "text-indigo-500" }
        ];

        const slider = document.getElementById('slider');
        let currentIndex = 0;
        let rotationCount = 0;
        let intervalId;

        // Initialize words in the DOM
        apps.forEach((app, index) => {
            const span = document.createElement('span');
            span.className = `word ${app.color}`;
            span.textContent = app.name;
            span.id = `word-${index}`;
            slider.appendChild(span);
        });

        function rotateWords() {
            const currentWord = document.getElementById(`word-${currentIndex}`);
            const nextIndex = (currentIndex + 1) % apps.length;
            const nextWord = document.getElementById(`word-${nextIndex}`);

            // Animate current out
            if (currentWord) {
                currentWord.classList.remove('active');
                currentWord.classList.add('exit');
                
                setTimeout(() => {
                    currentWord.classList.remove('exit');
                }, 500); 
            }

            // Animate next in
            if (nextWord) {
                nextWord.classList.remove('exit'); 
                requestAnimationFrame(() => {
                    nextWord.classList.add('active');
                });
            }

            currentIndex = nextIndex;

            // Stop condition: Back to "Hub"
            if (currentIndex === 0) {
                rotationCount++;
                if (rotationCount === 1) { // We completed one full loop
                    clearInterval(intervalId);
                    
                    // Wait a moment so "Hub" lingers, then trigger the outro
                    setTimeout(triggerOutro, 1500); 
                }
            }
        }

        function triggerOutro() {
            const brandText = document.getElementById('brand-text');
            const logoWrapper = document.getElementById('logo-wrapper');
            const logoSvg = document.getElementById('logo-svg');

            // 1. Shrink and fade text
            brandText.classList.add('outro-text');
            
            // 2. Remove margin from logo
            logoWrapper.classList.add('outro-wrapper');

            // 3. Morph logo into horizontal layout and trigger walking animation
            logoSvg.classList.add('outro-logo', 'walking');

            // 4. Trigger the wave after the walk completes (1.2s)
            setTimeout(() => {
                logoSvg.classList.remove('walking');
                logoSvg.classList.add('waving');
            }, 1200);
        }

        // Start initial sequence
        setTimeout(() => {
            const firstWord = document.getElementById('word-0');
            firstWord.classList.add('active');

            // Begin rotation loop
            intervalId = setInterval(rotateWords, 1800);
        }, 1600);

    </script>
