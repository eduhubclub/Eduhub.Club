  const monthlyClassProgress = 92; 
  
  const currentReports = useMemo(() => {
    const data = {};
    const today = todayRef.current;
    const daysInMonth = new Date(reportYear, reportMonth + 1, 0).getDate();

    for (let i = 1; i <= daysInMonth; i++) {
      const d = new Date(reportYear, reportMonth, i);
      if (d.getDay() === 0 || d.getDay() === 6) continue; // Skip weekends
      if (d > today) continue; // Skip future dates

      const total = STUDENT_DATABASE.length;
      // Using a pseudo-random approach based on date so it stays consistent between toggles
      const pseudoRandom1 = (reportYear + reportMonth + i) % 4; 
      const pseudoRandom2 = (reportYear + reportMonth + i) % 7;
      
      const absent = pseudoRandom1; 
      const present = total - absent;
      const homeLunch = pseudoRandom2 + 2; 
      const schoolLunch = present - homeLunch; 

      // Create consistent lists of students for the expanded view
      const offset = (reportYear + reportMonth + i) % total;
      const shiftedStudents = [...STUDENT_DATABASE.slice(offset), ...STUDENT_DATABASE.slice(0, offset)];
      
      const absentStudents = shiftedStudents.slice(0, absent);
      const presentStudents = shiftedStudents.slice(absent);
      const homeLunchStudents = presentStudents.slice(0, homeLunch);
      const schoolLunchStudents = presentStudents.slice(homeLunch);

      data[i] = {
        date: d,
        present,
        absent,
        schoolLunch,
        homeLunch,
        presentStudents,
        absentStudents,
        schoolLunchStudents,
        homeLunchStudents,
        isToday: d.getDate() === today.getDate() && d.getMonth() === today.getMonth() && d.getFullYear() === today.getFullYear()
      };
    }
    return data;
  }, [reportMonth, reportYear]);

    "District": { primary: "rose-500", bg: "bg-rose-500", text: "text-rose-500", border: "border-rose-500", hoverBg: "hover:bg-rose-50", ring: "ring-rose-500" },
    "Admin": { primary: "emerald-500", bg: "bg-emerald-500", text: "text-emerald-500", border: "border-emerald-500", hoverBg: "hover:bg-emerald-50", ring: "ring-emerald-500" },
    "Analytics": { primary: "orange-500", bg: "bg-orange-500", text: "text-orange-500", border: "border-orange-500", hoverBg: "hover:bg-orange-50", ring: "ring-orange-500" }

  const searchRef = useRef(null);
  const isLeft = sidebarSide === 'left';

  const toggleAttendance = (id, status) => {
    setAttendance(prev => ({
      ...prev,
      [id]: prev[id] === status ? null : status
    }));
  };

  const toggleLunch = (id, choice) => {
    setLunchChoices(prev => ({
      ...prev,
      [id]: prev[id] === choice ? null : choice
    }));
  };

    setExpandedReports(prev => ({
      ...prev,
      [dateStr]: !prev[dateStr]

    const unmarked = STUDENT_DATABASE.filter(s => !attendance[s.id]);

    const handleClickOutside = (event) => {
      if (popoutRef.current && !popoutRef.current.contains(event.target)) {
        setPopoutItem(null);
      }
      if (searchRef.current && !searchRef.current.contains(event.target) && !searchQuery) {
        setIsSearchExpanded(false);
      }
    };
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);

  const navHeight = 'h-16';

  const handleItemClick = (item) => {
    if (item.type === 'link') {
      setActiveTab(item.name);
      setExpandedItem(null);
      setPanelItem(null);
    } else if (item.type === 'accordion') {
      if (!isSidebarOpen) setIsSidebarOpen(true);
      setExpandedItem(expandedItem === item.id ? null : item.id);
      setPanelItem(null);
    } else if (item.type === 'popout') {
      setPopoutItem(popoutItem === item.id ? null : item.id);
    } else if (item.type === 'panel') {
      const isOpening = panelItem !== item.id;
      setPanelItem(isOpening ? item.id : null);
      setExpandedItem(null);
      if (isOpening) setActiveTab(null);
    }

  const renderDraggableStudent = (student) => (
    <div 
      key={student.id}
      draggable
      onDragStart={(e) => e.dataTransfer.setData('studentId', student.id.toString())}
      className={`flex items-center px-2 py-1.5 rounded-lg border shadow-sm cursor-grab active:cursor-grabbing transition-all hover:scale-105 ${isDarkMode ? 'bg-slate-800 border-slate-700 hover:bg-slate-700 text-white' : 'bg-slate-50 border-slate-200 hover:bg-slate-100 text-slate-800'}`}

      <span className={`text-[11px] font-semibold whitespace-nowrap pointer-events-none`}>
        {student.name}
      </span>
    </div>
  );

  const getStudentCardClass = (studentId, type) => {
    const base = `flex transition-all duration-200 group `;
    const mode = type === 'lunch' ? lunchViewMode : viewMode;
    
    let viewClass = '';
    if (mode === 'grid') {
      viewClass = 'flex-col items-center justify-center p-4 text-center aspect-square cursor-pointer border rounded-2xl ';
    } else {
      viewClass = 'flex-row items-center px-4 py-2 sm:px-6 sm:py-3 ';
    }
    
    let stateClass = '';
    if (type === 'lunch') {
      const choice = lunchChoices[studentId];
      if (choice === 'school') stateClass = isDarkMode ? 'bg-sky-900/30' : 'bg-sky-50';
      else if (choice === 'home') stateClass = isDarkMode ? 'bg-amber-900/30' : 'bg-amber-50';
      else stateClass = isDarkMode ? 'bg-transparent hover:bg-slate-800/50' : 'bg-transparent hover:bg-slate-50';
      
      if (mode === 'grid') {
        if (choice === 'school') stateClass += isDarkMode ? ' border-sky-500/30' : ' border-sky-200';
        else if (choice === 'home') stateClass += isDarkMode ? ' border-amber-500/30' : ' border-amber-200';
        else stateClass += isDarkMode ? ' border-slate-800 hover:border-slate-700' : ' border-slate-200 hover:border-slate-300 hover:shadow-md';
      }
    } else {
      const status = attendance[studentId];
      if (status === 'present') stateClass = isDarkMode ? 'bg-emerald-900/30' : 'bg-emerald-50';
      else if (status === 'absent') stateClass = isDarkMode ? 'bg-rose-900/30' : 'bg-rose-50';
      else stateClass = isDarkMode ? 'bg-transparent hover:bg-slate-800/50' : 'bg-transparent hover:bg-slate-50';

        if (status === 'present') stateClass += isDarkMode ? ' border-emerald-500/30' : ' border-emerald-200';
        else if (status === 'absent') stateClass += isDarkMode ? ' border-rose-500/30' : ' border-rose-200';
        else stateClass += isDarkMode ? ' border-slate-800 hover:border-slate-700' : ' border-slate-200 hover:border-slate-300 hover:shadow-md';
      }
    }
    return base + viewClass + stateClass;
  };

      if (lunchChoices[studentId] === 'school') return 'ring-2 ring-sky-500/50';
      if (lunchChoices[studentId] === 'home') return 'ring-2 ring-amber-500/50';
    } else {
      if (attendance[studentId] === 'present') return 'ring-2 ring-emerald-500/50';
      if (attendance[studentId] === 'absent') return 'ring-2 ring-rose-500/50';

    return '';

      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center mb-6 gap-4">

          <h1 className={`text-2xl font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Daily Attendance</h1>

            onClick={() => setViewMode('grid')}
            className={`flex items-center px-4 py-2 text-sm font-semibold rounded-md transition-all duration-200 ${viewMode === 'grid' ? (isDarkMode ? 'bg-slate-800 text-white shadow-sm' : 'bg-white text-slate-900 shadow-sm') : (isDarkMode ? 'text-slate-400 hover:text-slate-200' : 'text-slate-500 hover:text-slate-700')}`}

'bg-slate-800 text-white shadow-sm' : 'bg-white text-slate-900 shadow-sm') : (isDarkMode ? 'text-slate-400 hover:text-slate-200' : 'text-slate-500 hover:text-slate-700')}`}

'bg-slate-800 text-white shadow-sm' : 'bg-white text-slate-900 shadow-sm') : (isDarkMode ? 'text-slate-400 hover:text-slate-200' : 'text-slate-500 hover:text-slate-700')}`}

                    </div>
                  )}

            </div>

            <div 
              className={`flex-1 p-6 flex flex-col transition-colors ${isDarkMode ? 'hover:bg-slate-800/30' : 'hover:bg-slate-50/50'}`}
              onDragOver={(e) => e.preventDefault()}
              onDrop={(e) => {
                e.preventDefault();
                const id = e.dataTransfer.getData('studentId');
                if (id) setAttendance(prev => ({ ...prev, [id]: 'absent' }));
              }}
            >
              <div className={`flex items-center justify-center mb-4 pb-4 border-b-2 border-dashed ${isDarkMode ? 'border-slate-800' : 'border-slate-200'}`}>
                <X size={20} className={`mr-2 ${isDarkMode ? 'text-rose-400' : 'text-rose-500'}`} />

                  {STUDENT_DATABASE.filter(s => attendance[s.id] === 'absent').length}

                  </div>
                  <div>
                    <p className={`font-bold ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>No Winner Yet</p>
                    <p className={`text-xs ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Keep tracking attendance!</p>
                  </div>
                </div>
              )}
            </div>
          </div>

        </div>

        {/* Individual Student Goals */}
        <div>
          <h2 className={`text-lg font-bold mb-4 flex items-center ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>
            <Target size={20} className="mr-2" />
            Individual Student Goals
          </h2>
          
          <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-4">
            {STUDENT_DATABASE.map(student => {
              const current = studentProgress[student.id];
              const target = studentGoals[student.id];
              const progressPct = Math.min(100, (current / target) * 100);
              const isGoalMet = current >= target;

              return (
                <div key={student.id} className={`p-5 rounded-xl border flex flex-col transition-colors ${isDarkMode ? 'bg-slate-900 border-slate-800 hover:bg-slate-800/50' : 'bg-white border-slate-200 hover:border-slate-300 hover:shadow-sm'}`}>
                  <div className="flex items-center justify-between mb-4">
                    <div className="flex items-center">
                      <div className={`w-10 h-10 rounded-full flex items-center justify-center text-xl mr-3 ${isDarkMode ? 'bg-slate-800' : 'bg-slate-100'}`}>
                        {student.avatar}
                      </div>
                      <span className={`font-bold ${isDarkMode ? 'text-slate-200' : 'text-slate-800'}`}>{student.name}</span>
                    </div>
                    {isGoalMet && (
                      <div className="flex items-center justify-center w-8 h-8 rounded-full bg-amber-100 text-amber-500 animate-in zoom-in duration-300" title="Goal Met!">
                        <Star size={16} className="fill-current" />
                      </div>
                    )}
                  </div>
                  
                  <div className="flex items-center justify-between mb-2">
                    <span className={`text-[10px] sm:text-xs font-semibold uppercase tracking-wider ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Perfect Days</span>
                    <div className="flex items-center">
                      <span className={`text-sm font-bold mr-1 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>{current} /</span>
                      <input 
                        type="number"
                        value={target}
                        onChange={(e) => setStudentGoals(prev => ({...prev, [student.id]: Number(e.target.value)}))}
                        className={`w-9 bg-transparent text-sm font-bold outline-none border-b text-center ${isDarkMode ? 'border-slate-700 text-white focus:border-purple-500' : 'border-slate-300 text-slate-900 focus:border-purple-500'}`}
                        min="1" max="31"
                      />
                    </div>
                  </div>
                  
                  <div className={`h-2 w-full rounded-full overflow-hidden mt-1 ${isDarkMode ? 'bg-slate-800' : 'bg-slate-100'}`}>
                    <div 
                      className={`h-full rounded-full transition-all duration-1000 ease-out ${isGoalMet ? 'bg-amber-400' : theme.bg}`}
                      style={{ width: `${progressPct}%` }}
                    ></div>
                  </div>
                </div>
              );
            })}
          </div>
        </div>

      </div>
    );
  };

  return (
    <div className={`flex min-h-screen font-sans transition-all duration-300 ${isDarkMode ? 'bg-slate-950 text-slate-100' : 'bg-slate-50 text-slate-900'} ${isLeft ? 'flex-row' : 'flex-row-reverse'}`}>
      
      {/* 1. PRIMARY SIDEBAR */}
      <aside className={`relative z-50 transition-all duration-300 ease-in-out flex flex-col shadow-sm ${sidebarWidth} ${isLeft ? 'border-r' : 'border-l'} ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
        <div className={`flex items-center justify-center shrink-0 border-b ${navHeight} w-full ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
          <div className="flex items-center justify-center px-4">
            {isSidebarOpen ? <LogoHorizontal className="h-6 w-auto" /> : <LogoIcon2x2 className="w-8 h-8" />}
          </div>
        </div>

        <div className="pt-2 shrink-0"></div>

        <nav className="flex-1 px-2 space-y-1 pt-2 pb-10 relative overflow-x-visible">
          {navigationItems.map((item) => {
            const isTabActive = activeTab === item.name || (item.type === 'accordion' && item.subItems.includes(activeTab));
            const isPanelOpen = panelItem === item.id;
            const isVisuallyActive = isTabActive || isPanelOpen;
            const isExpanded = expandedItem === item.id;
            const hasPopout = popoutItem === item.id;

            return (
              <div key={item.id} className="relative">
                <button
                  onClick={() => handleItemClick(item)}
                  className={`flex items-center transition-all group relative rounded-xl h-12
                    ${isSidebarOpen ? 'w-full px-3' : 'w-16 mx-auto justify-center'}
                    ${isVisuallyActive ? `${theme.bg} text-white shadow-lg` : `${isDarkMode ? 'text-slate-400 hover:bg-slate-800 hover:text-slate-200' : 'text-slate-500 hover:bg-slate-100 hover:text-slate-900'}`}`}
                >
                  <div className={`flex items-center justify-center shrink-0`}>
                    <item.icon size={20} strokeWidth={isVisuallyActive ? 2.5 : 2} />
                  </div>
                  {isSidebarOpen && (
                    <>
                      <span className="ml-4 font-semibold text-sm flex-1 text-left truncate">{item.name}</span>
                      {item.type === 'accordion' && (
                        <ChevronDown size={14} className={`transition-transform duration-200 shrink-0 ${isExpanded ? 'rotate-180' : ''}`} />
                      )}
                      {item.type === 'panel' && (
                        <ChevronRight size={14} className={`transition-all duration-200 shrink-0 ${isLeft ? '' : 'rotate-180'} ${isPanelOpen ? 'opacity-100' : 'opacity-0 group-hover:opacity-40'}`} />
                      )}
                      {item.type === 'popout' && <MoreVertical size={14} className="opacity-40 shrink-0" />}
                    </>
                  )}
                  {!isSidebarOpen && (
                    <div className={`absolute ${isLeft ? 'left-full ml-4' : 'right-full mr-4'} px-3 py-1.5 bg-slate-900 text-white text-xs font-bold rounded-md opacity-0 group-hover:opacity-100 pointer-events-none transition-opacity z-[60] whitespace-nowrap shadow-xl`}>
                      {item.name}
                    </div>
                  )}
                </button>
                {item.type === 'accordion' && isExpanded && isSidebarOpen && (
                  <div className={`mt-1 space-y-1 ${isDarkMode ? 'border-slate-800' : 'border-slate-100'} ${isLeft ? 'ml-10 border-l-2' : 'mr-10 border-r-2'}`}>
                    {item.subItems.map(sub => (
                      <button 
                        key={sub}
                        onClick={() => setActiveTab(sub)}
                        className={`block w-full text-left py-2 px-4 text-xs font-medium transition-colors
                          ${isLeft ? 'rounded-r-lg' : 'rounded-l-lg text-right'}
                          ${activeTab === sub ? `${theme.text} ${isDarkMode ? 'bg-slate-800' : 'bg-slate-50'} font-bold` : `${isDarkMode ? 'text-slate-500 hover:text-slate-300 hover:bg-slate-800/50' : 'text-slate-400 hover:text-slate-600 hover:bg-slate-50/50'}`}`}
                      >
                        {sub}
                      </button>
                    ))}
                  </div>
                )}
                {item.type === 'popout' && hasPopout && (
                  <div 
                    ref={popoutRef} 
                    className={`absolute z-[100] border rounded-xl shadow-[0_20px_50px_rgba(0,0,0,0.15)] py-2 min-w-[200px]
                      ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}
                      ${isLeft ? 'left-full ml-4' : 'right-full mr-4'}
                      top-0`}
                  >
                    <div className={`px-3 py-2 mb-1 border-b ${isDarkMode ? 'border-slate-800/50' : 'border-slate-50'}`}>
                      <p className="text-[10px] font-bold text-slate-400 uppercase tracking-widest">{item.name} Options</p>
                    </div>
                    {item.actions.map(action => (
                      <button
                        key={action.name}
                        className={`w-full flex items-center px-3 py-2.5 text-xs font-semibold transition-colors ${action.color || (isDarkMode ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-600 hover:bg-slate-50')}`}
                        onClick={() => { setPopoutItem(null); setActiveTab(`${item.name}: ${action.name}`); }}
                      >
                        <div className={`w-6 h-6 rounded flex items-center justify-center mr-3 ${action.color ? 'bg-rose-50' : (isDarkMode ? 'bg-slate-800' : 'bg-slate-50')}`}>
                           <action.icon size={12} />
                        </div>
                        {action.name}
                      </button>
                    ))}
                  </div>
                )}
              </div>
            );
          })}
        </nav>

        {/* BOTTOM QUICK SETTINGS SECTION */}
        <div className={`p-2 border-t ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
           <div className={`flex transition-all duration-300 ${isSidebarOpen ? 'flex-row items-center space-x-2' : 'flex-col items-center space-y-2'}`}>
              
              <button 
                onClick={() => setIsDarkMode(!isDarkMode)}
                className={`flex items-center justify-center rounded-xl transition-all h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'w-16'} ${isDarkMode ? 'text-amber-400 bg-slate-800 hover:bg-slate-700' : 'text-slate-400 bg-slate-50 hover:bg-slate-100 hover:text-slate-900'}`}
                title={isDarkMode ? "Switch to Light Mode" : "Switch to Dark Mode"}
              >
                <div className="shrink-0 flex items-center justify-center">
                  {isDarkMode ? <Sun size={18} /> : <Moon size={18} />}
                </div>
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Dark</span>}
              </button>

              <button 
                onClick={() => { setSidebarSide(isLeft ? 'right' : 'left'); setPanelItem(null); }}
                className={`flex items-center justify-center rounded-xl transition-all h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'w-16'} ${isDarkMode ? 'text-slate-500 hover:bg-slate-800 hover:text-slate-200' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}
                title="Flip Sidebar Side"
              >
                <div className="shrink-0 flex items-center justify-center">
                  <ArrowLeftRight size={16} />
                </div>
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Flip</span>}
              </button>

              <button
                onClick={() => setIsSidebarOpen(!isSidebarOpen)}
                className={`flex items-center justify-center transition-all rounded-xl h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'mx-auto w-16'} ${isDarkMode ? 'text-slate-400 bg-slate-800/40 hover:bg-slate-700/60 hover:text-slate-200' : 'text-slate-400 bg-slate-50 hover:bg-slate-100 hover:text-slate-900'}`}
                title={isSidebarOpen ? "Collapse Sidebar" : "Expand Sidebar"}
              >
                <div className="shrink-0 flex items-center justify-center">
                  {isSidebarOpen ? <ChevronLeft size={18} className={!isLeft ? 'rotate-180' : ''} /> : <Menu size={20} />}
                </div>
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Menu</span>}
              </button>
           </div>
        </div>
      </aside>

      {/* 2. SECONDARY SIDEBAR PANEL */}
      <div 
        className={`z-40 transition-all duration-300 ease-in-out overflow-hidden flex flex-col
          ${panelItem ? 'w-72' : 'w-0'}
          ${isDarkMode ? 'bg-slate-900' : 'bg-white'}
          ${isLeft ? `border-r ${isDarkMode ? 'border-slate-800' : 'border-slate-200'}` : `border-l ${isDarkMode ? 'border-slate-800' : 'border-slate-200'}`}`}
      >
        <div className={`w-72 flex flex-col h-full transition-opacity duration-300 ${panelItem ? 'opacity-100' : 'opacity-0'}`}>
          <div className={`flex items-center justify-between p-6 shrink-0 border-b ${isDarkMode ? 'border-slate-800' : 'border-slate-100'} ${navHeight}`}>
            <h2 className="text-[10px] font-black text-slate-400 uppercase tracking-[0.2em]">Resources</h2>
            <button onClick={() => setPanelItem(null)} className={`p-1 rounded-md transition-colors ${isDarkMode ? 'text-slate-400 hover:text-white hover:bg-slate-800' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-50'}`}>
              <X size={16} />
            </button>
          </div>
          <div className="flex-1 overflow-y-auto">
            {navigationItems.find(i => i.id === panelItem)?.panelContent?.map((sub, index, arr) => (
              <button 
                key={sub.label}
                onClick={() => setActiveTab(`Resource: ${sub.label}`)}
                className={`w-full flex items-center p-5 text-left group transition-all relative
                  ${isDarkMode ? 'hover:bg-slate-800/50' : 'hover:bg-slate-50/50'}
                  ${index !== arr.length - 1 ? `border-b ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}` : ''}`}
              >
                <div className={`w-10 h-10 rounded-xl flex items-center justify-center transition-all duration-300 ${isDarkMode ? 'bg-slate-800 border-slate-700 text-slate-400 group-hover:text-white group-hover:bg-slate-700' : 'bg-slate-50 border border-slate-100 text-slate-400 group-hover:bg-white group-hover:shadow-sm'} group-hover:${theme.text} group-hover:${theme.border}`}>
                  <sub.icon size={18} />
                </div>
                <div className="ml-4 flex-1">
                  <span className={`block text-sm font-bold group-hover:${theme.text} transition-colors ${isDarkMode ? 'text-slate-200' : 'text-slate-700'}`}>{sub.label}</span>
                  <span className="block text-[10px] text-slate-500 font-medium uppercase tracking-tight">Access files</span>
                </div>
                <ChevronRight size={14} className={`text-slate-300 opacity-0 group-hover:opacity-100 transition-all transform ${isLeft ? 'group-hover:translate-x-1' : 'rotate-180 group-hover:-translate-x-1'} group-hover:${theme.text}`} />
              </button>
            ))}
          </div>
          <div className={`p-4 border-t ${isDarkMode ? 'bg-slate-900/50 border-slate-800' : 'bg-slate-50/50 border-slate-100'}`}>
            <button className={`w-full py-3 px-4 rounded-xl border-2 border-dashed text-xs font-bold transition-all flex items-center justify-center ${isDarkMode ? 'border-slate-800 text-slate-500 hover:border-slate-700 hover:text-slate-300' : 'border-slate-200 text-slate-400 hover:border-slate-300 hover:text-slate-600'} hover:${theme.border} hover:${theme.text}`}>
               <Plus size={14} className="mr-2" /> Create New
            </button>
          </div>
        </div>
      </div>

      {/* MAIN CONTENT AREA */}
      <div className="flex-1 flex flex-col min-w-0 relative z-10">
        <header className={`border-b flex items-center px-6 z-20 shrink-0 ${navHeight} sticky top-0 transition-colors duration-300 ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
          
          <div className={`transition-all duration-500 overflow-hidden flex items-center shrink-0 ${isSearchExpanded && !useOverlaySearch ? 'w-0 opacity-0 mr-0' : 'flex-1 opacity-100 mr-4'}`}>
            <div className="flex items-center text-lg font-bold whitespace-nowrap tracking-tight">
              <span className={isDarkMode ? 'text-white' : 'text-slate-900'}>Edu.</span>
              <span className={theme.text}>{currentAppName}</span>
            </div>
          </div>
          
          {useOverlaySearch && isSearchExpanded && (
            <div className={`absolute inset-x-0 inset-y-0 z-50 flex items-center px-4 animate-in fade-in slide-in-from-top-2 duration-300 ${isDarkMode ? 'bg-slate-900' : 'bg-white'}`}>
                <Search size={18} className="text-slate-400 mr-3" />
                <input
                    type="text"
                    placeholder="Search..."
                    autoFocus
                    className={`flex-1 bg-transparent border-none outline-none text-sm font-medium ${isDarkMode ? 'text-white' : 'text-slate-900'}`}
                    value={searchQuery}
                    onChange={(e) => setSearchQuery(e.target.value)}
                />
                <button onClick={() => setIsSearchExpanded(false)} className="p-2 text-slate-400 hover:text-slate-900">
                    <X size={18} />
                </button>
            </div>
          )}
          
          <div className={`flex items-center justify-end flex-1 h-full relative`}>
             {!useOverlaySearch && (
               <div 
                  ref={searchRef} 
                  className={`flex items-center relative transition-all duration-500 ease-in-out h-10 justify-end
                    ${isSearchExpanded ? 'flex-1 max-w-xl mr-4' : 'w-10'}`}
               >
                  <div className={`relative h-full transition-all duration-500 overflow-hidden flex items-center ${isSearchExpanded ? 'w-full' : 'w-0'}`}>
                    <div className={`absolute left-0 flex items-center justify-center w-12 h-10 z-30 ${isSearchExpanded ? theme.text : 'text-slate-400'}`}>
                        <Search size={18} />
                    </div>

                    <input
                      type="text"
                      placeholder="Search resources..."
                      value={searchQuery}
                      onChange={(e) => setSearchQuery(e.target.value)}
                      className={`w-full h-full border rounded-full pl-12 pr-12 text-sm font-medium outline-none focus:outline-none transition-all duration-300 ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white focus:bg-slate-700 focus:border-slate-600' : 'bg-slate-50 border-slate-200 text-slate-900 focus:bg-white focus:border-slate-300'}`}
                      autoFocus={isSearchExpanded}
                    />
                    
                    <div className="absolute right-0 flex items-center justify-center w-12 h-10 z-30">
                        <button 
                            onClick={(e) => { 
                                e.stopPropagation();
                                if (searchQuery) {
                                    setSearchQuery("");
                                } else {
                                    setIsSearchExpanded(false);
                                }
                            }}
                            className={`p-1.5 text-slate-400 hover:text-slate-900 transition-all duration-300 transform
                            ${isSearchExpanded ? 'opacity-100 scale-100' : 'opacity-0 scale-50 pointer-events-none'}`}
                        >
                            <X size={16} strokeWidth={2.5} />
                        </button>
                    </div>
                  </div>

                  <button 
                    onClick={() => setIsSearchExpanded(true)}
                    className={`absolute right-0 p-2 text-slate-400 hover:${theme.text} rounded-full transition-all duration-300 z-20 
                      ${isSearchExpanded ? 'opacity-0 pointer-events-none' : 'opacity-100'}`}
                  >
                    <Search size={18} />
                  </button>
               </div>
             )}

             {useOverlaySearch && !isSearchExpanded && (
                <button onClick={() => setIsSearchExpanded(true)} className="p-2 text-slate-400 hover:text-slate-900 rounded-full mr-2">
                    <Search size={18} />
                </button>
             )}
             
             <div className={`flex items-center space-x-2 shrink-0 transition-all duration-300 ${isSearchExpanded && useOverlaySearch ? 'opacity-0' : 'opacity-100'}`}>
                <button className={`p-2 text-slate-400 hover:${theme.text} rounded-full transition-colors`}>
                  <Grid size={18} />
                </button>
                <button className={`p-2 text-slate-400 hover:${theme.text} rounded-full transition-colors relative`}>
                  <Bell size={18} />
                  <span className={`absolute top-2.5 right-2.5 w-1.5 h-1.5 ${theme.bg} rounded-full border border-white`}></span>
                </button>
                <div className={`h-6 w-[1px] mx-1 shrink-0 ${isDarkMode ? 'bg-slate-800' : 'bg-slate-200'}`}></div>
                <button className={`w-9 h-9 rounded-full flex items-center justify-center transition-all shadow-sm ${isDarkMode ? 'bg-slate-800 border-slate-700 text-slate-400 hover:text-white hover:border-slate-600' : 'bg-slate-50 border-slate-200 text-slate-400 hover:bg-white hover:shadow-md'}`}>
                  <User size={20} strokeWidth={2} />
                </button>
             </div>
          </div>
        </header>

        <main className="flex-1 overflow-auto p-10 relative">
          {activeTab === 'Daily Attendance' && renderDailyAttendance()}
          {activeTab === 'Lunch Count' && renderLunchCount()}
          {activeTab === 'Reports' && renderReports()}
          {activeTab === 'Awards' && renderAwards()}
        </main>
      </div>
    </div>
  );
};

export default App;
