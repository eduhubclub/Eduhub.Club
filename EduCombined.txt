const getStudentAvatar = (student) => {
  if (!student) return '';

  if (student.avatar) return student.avatar;

  const handleEmojiSelect = (emoji) => {
    const updatedStudent = { ...selectedClassStudent, emoji };
    setSelectedClassStudent(updatedStudent);

    const updatedClasses = classes.map(cls => {
      if (selectedClass && cls.id === selectedClass.id) {
        return {
          ...cls,
          studentList: cls.studentList.map(s => 
            (s.name === selectedClassStudent.name && s.id === selectedClassStudent.id) ? updatedStudent : s
          )
        };
      }
      return cls;
    });
    
    setClasses(updatedClasses);
    if (selectedClass) {
      setSelectedClass(updatedClasses.find(c => c.id === selectedClass.id));
    }
    setIsEmojiPickerOpen(false);
  };

  // Randomizer App Logic
  const pickRandomStudent = () => {
    if (isAnimating || activeStudents.length === 0) return;
    setIsAnimating(true);
    
    let shuffles = 0;
    const maxShuffles = 12;
    const interval = setInterval(() => {
      const randomIndex = Math.floor(Math.random() * activeStudents.length);
      setSelectedRandomStudent(activeStudents[randomIndex]);
      shuffles++;
      
      if (shuffles >= maxShuffles) {
        clearInterval(interval);
        setIsAnimating(false);
      }
    }, 75);
  };

  const spinWheel = () => {
    if (isSpinningWheel || activeStudents.length === 0) return;
    setIsSpinningWheel(true);
    setWheelWinner(null);

    const spinSpins = 5 + Math.random() * 5; 
    const extraDegrees = Math.random() * 360;
    const totalRotation = wheelRotation + (spinSpins * 360) + extraDegrees;

    setWheelRotation(totalRotation);

    const normalizedRotation = totalRotation % 360;
    const step = 360 / activeStudents.length;
    const pointerAngle = (360 - normalizedRotation + 90) % 360;
    const winningIndex = Math.floor(pointerAngle / step);

    setTimeout(() => {
      setIsSpinningWheel(false);
      setWheelWinner(activeStudents[winningIndex]);
    }, 4000);
  };

  const handleShuffleDeck = () => {
    setShuffledDeck([...activeStudents].sort(() => Math.random() - 0.5));
    setFlippedCards(new Set());
  };

  const onPullDragStart = (e) => {
    if (hasPulledName) return;
    setIsPullDragging(true);
    const clientY = e.touches ? e.touches[0].clientY : e.clientY;
    setPullStartY(clientY - pullDragY);
  };

  const onPullDragMove = (e) => {
    if (!isPullDragging || hasPulledName) return;
    const clientY = e.touches ? e.touches[0].clientY : e.clientY;
    let newY = clientY - pullStartY;
    
    if (newY > 0) newY = 0; 
    
    setPullDragY(newY);

    if (newY < -250) {
      setHasPulledName(true);
      setIsPullDragging(false);
      setPullDragY(-600);
    }
  };

  const onPullDragEnd = () => {
    if (!isPullDragging || hasPulledName) return;
    setIsPullDragging(false);
    
    if (pullDragY > -250) {
      setPullDragY(0); 
    }
  };

  const resetPull = () => {
    setHasPulledName(false);
    setPullDragY(0);
    setStagedStudent(null); 
  };

  const handlePullRemove = () => {
    setActiveStudents(prev => prev.filter(s => s.id !== stagedStudent.id));
    resetPull();
  };

  const renderActiveStudentsPopout = () => {
    const baseStudents = randomizerSelectedClass?.studentList || STUDENT_DATABASE;
    if (activeStudents.length === baseStudents.length) return null;

    return (
      <div className="absolute top-0 right-0 z-40 flex flex-col items-end" ref={activeListRef}>
        <button 
          onClick={() => setShowActiveList(!showActiveList)} 
          className={`px-4 py-2 flex items-center space-x-1.5 rounded-full font-bold text-xs transition-all shadow-sm ${isDarkMode ? 'bg-slate-800 text-slate-300 hover:bg-slate-700' : 'bg-white text-slate-600 hover:bg-slate-50 border border-slate-200'}`}
        >
          <span>{activeStudents.length} / {baseStudents.length} Active</span>
          <ChevronDown size={14} className={`transition-transform duration-200 ${showActiveList ? 'rotate-180' : ''}`} />
        </button>

        {showActiveList && (
          <div className={`mt-2 w-64 max-h-[50vh] overflow-y-auto rounded-xl shadow-2xl border flex flex-col text-left animate-in fade-in slide-in-from-top-2 duration-200 ${isDarkMode ? 'bg-slate-800 border-slate-700 text-slate-200' : 'bg-white border-slate-200 text-slate-800'}`}>
            <div className={`p-3 border-b flex justify-between items-center sticky top-0 z-10 ${isDarkMode ? 'bg-slate-800/95 border-slate-700 backdrop-blur-sm' : 'bg-white/95 border-slate-100 backdrop-blur-sm'}`}>
              <span className="font-bold text-sm">Active Students</span>
              <button 
                onClick={() => { setActiveStudents(baseStudents); setShowActiveList(false); }} 
                className={`text-xs font-bold ${theme.text} hover:underline`}
              >
                Reset All
              </button>
            </div>
            <div className="p-2 flex flex-col space-y-1">
              {activeStudents.map((s, i) => (
                <div key={s.id || `active-student-${i}`} className={`flex items-center space-x-3 p-2 rounded-lg transition-colors ${isDarkMode ? 'hover:bg-slate-700' : 'hover:bg-slate-50'}`}>
                  <span className="text-lg leading-none w-8 h-8 flex items-center justify-center bg-slate-100 dark:bg-slate-700 rounded-full text-sm">{getStudentAvatar(s)}</span>
                  <span className="text-sm font-semibold truncate">{s.name}</span>
                </div>
              ))}
              {activeStudents.length === 0 && (
                <div className="p-4 text-center text-sm opacity-50 font-medium">All students removed.</div>
              )}
            </div>
          </div>
        )}
      </div>
    );
  };

  const contextValue = { 
    classes, setClasses, 
    selectedClass, setSelectedClass, 
    selectedStudent: selectedClassStudent, 
    setSelectedStudent: setSelectedClassStudent 
  };

  return (
    <ClassContext.Provider value={contextValue}>
      <div className={`flex min-h-screen font-sans transition-all duration-300 ${isDarkMode ? 'bg-slate-950 text-slate-100' : 'bg-slate-50 text-slate-900'} ${isLeft ? 'flex-row' : 'flex-row-reverse'}`}>
      
        {/* 1. PRIMARY SIDEBAR */}
        <aside className={`relative z-50 transition-all duration-300 ease-in-out flex flex-col shadow-sm ${sidebarWidth} ${isLeft ? 'border-r' : 'border-l'} ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
          <div className={`flex items-center justify-center shrink-0 border-b ${navHeight} w-full ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
            <div className="flex items-center justify-center px-4">
              {isSidebarOpen ? <LogoHorizontal className="h-6 w-auto" /> : <LogoIcon2x2 className="w-8 h-8" />}
            </div>
          </div>

        <div className="pt-2 shrink-0"></div>

        <nav className="flex-1 px-2 space-y-1 pt-2 pb-10 relative overflow-x-visible">
          {navigationItems.map((item) => {
            const isTabActive = activeTab === item.name || (item.type === 'accordion' && item.subItems?.includes(activeTab));
            const isPanelOpen = panelItem === item.id;
            const isVisuallyActive = isTabActive || isPanelOpen;
            const isExpanded = expandedItem === item.id;
            const hasPopout = popoutItem === item.id;

            return (
              <div key={item.id} className="relative">
                <button
                  onClick={() => handleItemClick(item)}
                  className={`flex items-center transition-all group relative rounded-xl h-12
                    ${isSidebarOpen ? 'w-full px-3' : 'w-16 mx-auto justify-center'}
                    ${isVisuallyActive ? `${theme.bg} text-white shadow-lg` : `${isDarkMode ? 'text-slate-400 hover:bg-slate-800 hover:text-slate-200' : 'text-slate-500 hover:bg-slate-100 hover:text-slate-900'}`}`}
                >
                  <div className={`flex items-center justify-center shrink-0`}>
                    <item.icon size={20} strokeWidth={isVisuallyActive ? 2.5 : 2} />
                  </div>
                  {isSidebarOpen && (
                    <>
                      <span className="ml-4 font-semibold text-sm flex-1 text-left truncate">{item.name}</span>
                      {item.type === 'accordion' && (
                        <ChevronDown size={14} className={`transition-transform duration-200 shrink-0 ${isExpanded ? 'rotate-180' : ''}`} />
                      )}
                      {item.type === 'panel' && (
                        <ChevronRight size={14} className={`transition-all duration-200 shrink-0 ${isLeft ? '' : 'rotate-180'} ${isPanelOpen ? 'opacity-100' : 'opacity-0 group-hover:opacity-40'}`} />
                      )}
                      {item.type === 'popout' && <MoreVertical size={14} className="opacity-40 shrink-0" />}
                    </>
                  )}
                  {!isSidebarOpen && (
                    <div className={`absolute ${isLeft ? 'left-full ml-4' : 'right-full mr-4'} px-3 py-1.5 bg-slate-900 text-white text-xs font-bold rounded-md opacity-0 group-hover:opacity-100 pointer-events-none transition-opacity z-[60] whitespace-nowrap shadow-xl`}>
                      {item.name}
                    </div>
                  )}
                </button>
                {item.type === 'accordion' && isExpanded && isSidebarOpen && item.subItems && (
                  <div className={`mt-1 space-y-1 ${isDarkMode ? 'border-slate-800' : 'border-slate-100'} ${isLeft ? 'ml-10 border-l-2' : 'mr-10 border-r-2'}`}>
                    {item.subItems.map(sub => (
                      <button 
                        key={sub}
                        onClick={() => {
                          if (activeApp === 'Classes') setClassesTab(sub);
                          else setRandomizerTab(sub);
                        }}
                        className={`block w-full text-left py-2 px-4 text-xs font-medium transition-colors
                          ${isLeft ? 'rounded-r-lg' : 'rounded-l-lg text-right'}
                          ${activeTab === sub ? `${theme.text} ${isDarkMode ? 'bg-slate-800' : 'bg-slate-50'} font-bold` : `${isDarkMode ? 'text-slate-500 hover:text-slate-300 hover:bg-slate-800/50' : 'text-slate-400 hover:text-slate-600 hover:bg-slate-50/50'}`}`}
                      >
                        {sub}
                      </button>
                    ))}
                  </div>
                )}
                {item.type === 'popout' && hasPopout && item.actions && (
                  <div 
                    ref={popoutRef} 
                    className={`absolute z-[100] border rounded-xl shadow-[0_20px_50px_rgba(0,0,0,0.15)] py-2 min-w-[200px]
                      ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}
                      ${isLeft ? 'left-full ml-4' : 'right-full mr-4'}
                      top-0`}
                  >
                    <div className={`px-3 py-2 mb-1 border-b ${isDarkMode ? 'border-slate-800/50' : 'border-slate-50'}`}>
                      <p className="text-[10px] font-bold text-slate-400 uppercase tracking-widest">{item.name} Options</p>
                    </div>
                    {item.actions.map(action => (
                      <button
                        key={action.name}
                        className={`w-full flex items-center px-3 py-2.5 text-xs font-semibold transition-colors ${action.color || (isDarkMode ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-600 hover:bg-slate-50')}`}
                        onClick={() => { 
                          setPopoutItem(null); 
                          if (activeApp === 'Classes') setClassesTab(`${item.name}: ${action.name}`);
                          else setRandomizerTab(`${item.name}: ${action.name}`);
                        }}
                      >
                        <div className={`w-6 h-6 rounded flex items-center justify-center mr-3 ${action.color ? 'bg-rose-50' : (isDarkMode ? 'bg-slate-800' : 'bg-slate-50')}`}>
                           <action.icon size={12} />
                        </div>
                        {action.name}
                      </button>
                    ))}
                  </div>
                )}
              </div>
            );
          })}
        </nav>

        {/* BOTTOM QUICK SETTINGS SECTION */}
        <div className={`p-2 border-t ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
           <div className={`flex transition-all duration-300 ${isSidebarOpen ? 'flex-row items-center space-x-2' : 'flex-col items-center space-y-2'}`}>
              
              {/* DARK MODE - LEFT / TOP */}
              <button 
                onClick={() => setIsDarkMode(!isDarkMode)}
                className={`flex items-center justify-center rounded-xl transition-all h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'w-16'} ${isDarkMode ? 'text-amber-400 bg-slate-800 hover:bg-slate-700' : 'text-slate-400 bg-slate-50 hover:bg-slate-100 hover:text-slate-900'}`}
                title={isDarkMode ? "Switch to Light Mode" : "Switch to Dark Mode"}
              >
                <div className="shrink-0 flex items-center justify-center">
                  {isDarkMode ? <Sun size={18} /> : <Moon size={18} />}
                </div>
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Dark</span>}
              </button>

              {/* FLIP - MIDDLE */}
              <button 
                onClick={() => { setSidebarSide(isLeft ? 'right' : 'left'); setPanelItem(null); }}
                className={`flex items-center justify-center rounded-xl transition-all h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'w-16'} ${isDarkMode ? 'text-slate-500 hover:bg-slate-800 hover:text-slate-200' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}
                title="Flip Sidebar Side"
              >
                <div className="shrink-0 flex items-center justify-center">
                  <ArrowLeftRight size={16} />
                </div>
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Flip</span>}
              </button>

              {/* COLLAPSE - RIGHT / BOTTOM */}
              <button
                onClick={() => setIsSidebarOpen(!isSidebarOpen)}
                className={`flex items-center justify-center transition-all rounded-xl h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'mx-auto w-16'} ${isDarkMode ? 'text-slate-400 bg-slate-800/40 hover:bg-slate-700/60 hover:text-slate-200' : 'text-slate-400 bg-slate-50 hover:bg-slate-100 hover:text-slate-900'}`}
                title={isSidebarOpen ? "Collapse Sidebar" : "Expand Sidebar"}
              >
                <div className="shrink-0 flex items-center justify-center">
                  {isSidebarOpen ? <ChevronLeft size={18} className={!isLeft ? 'rotate-180' : ''} /> : <Menu size={20} />}
                </div>
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Menu</span>}
              </button>
           </div>
        </div>
      </aside>

      {/* 2. SECONDARY SIDEBAR PANEL */}
      <div 
        className={`z-40 transition-all duration-300 ease-in-out overflow-hidden flex flex-col
          ${panelItem ? 'w-72' : 'w-0'}
          ${isDarkMode ? 'bg-slate-900' : 'bg-white'}
          ${isLeft ? `border-r ${isDarkMode ? 'border-slate-800' : 'border-slate-200'}` : `border-l ${isDarkMode ? 'border-slate-800' : 'border-slate-200'}`}`}
      >
        <div className={`w-72 flex flex-col h-full transition-opacity duration-300 ${panelItem ? 'opacity-100' : 'opacity-0'}`}>
          <div className={`flex items-center justify-between p-6 shrink-0 border-b ${isDarkMode ? 'border-slate-800' : 'border-slate-100'} ${navHeight}`}>
            <h2 className="text-[10px] font-black text-slate-400 uppercase tracking-[0.2em]">
              {panelItem === 'classes-panel' ? 'Select Class' : 'Resources'}
            </h2>
            <button onClick={() => setPanelItem(null)} className={`p-1 rounded-md transition-colors ${isDarkMode ? 'text-slate-400 hover:text-white hover:bg-slate-800' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-50'}`}>
              <X size={16} />
            </button>
          </div>
          <div className="flex-1 overflow-y-auto">
            {panelItem === 'classes-panel' ? (
              classes.map((cls) => {
                const ClassIcon = iconMap[cls.icon] || BookOpen;
                const isSelected = randomizerSelectedClass?.id === cls.id;
                return (
                  <button 
                    key={cls.id}
                    onClick={() => {
                      setRandomizerSelectedClass(cls);
                      setActiveStudents(cls.studentList || []);
                      setSelectedRandomStudent(null);
                      setWheelWinner(null);
                      setHasPulledName(false);
                      setStagedStudent(null);
                      setShuffledDeck([...(cls.studentList || [])].sort(() => Math.random() - 0.5));
                      setFlippedCards(new Set());
                      setPanelItem(null); 
                    }}
                    className={`w-full flex items-center p-5 text-left group transition-all relative
                      ${isDarkMode ? 'hover:bg-slate-800/50' : 'hover:bg-slate-50/50'}
                      ${isSelected ? (isDarkMode ? 'bg-slate-800' : 'bg-slate-100') : ''}
                      border-b ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}
                  >
                    <div className={`w-10 h-10 rounded-xl flex items-center justify-center transition-all duration-300 ${isSelected ? 'bg-rose-500 text-white shadow-md' : isDarkMode ? 'bg-slate-800 border-slate-700 text-slate-400 group-hover:text-white group-hover:bg-slate-700' : 'bg-slate-50 border border-slate-100 text-slate-400 group-hover:bg-white group-hover:shadow-sm'}`}>
                      <ClassIcon size={18} />
                    </div>
                    <div className="ml-4 flex-1">
                      <span className={`block text-sm font-bold transition-colors ${isSelected ? (isDarkMode ? 'text-white' : 'text-slate-900') : (isDarkMode ? 'text-slate-200' : 'text-slate-700')}`}>{cls.name}</span>
                      <span className="block text-[10px] text-slate-500 font-medium uppercase tracking-tight">{cls.studentList?.length || 0} Students</span>
                    </div>
                    {isSelected && <CheckCircle size={16} className="text-rose-500" />}
                  </button>
                )
              })
            ) : (
              navigationItems.find(i => i.id === panelItem)?.panelContent?.map((sub, index, arr) => (
                <button 
                  key={sub.label}
                  onClick={() => {
                    if (activeApp === 'Classes') setClassesTab(`Resource: ${sub.label}`);
                    else setRandomizerTab(`Resource: ${sub.label}`);
                  }}
                  className={`w-full flex items-center p-5 text-left group transition-all relative
                    ${isDarkMode ? 'hover:bg-slate-800/50' : 'hover:bg-slate-50/50'}
                    ${index !== arr.length - 1 ? `border-b ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}` : ''}`}
                >
                  <div className={`w-10 h-10 rounded-xl flex items-center justify-center transition-all duration-300 ${isDarkMode ? 'bg-slate-800 border-slate-700 text-slate-400 group-hover:text-white group-hover:bg-slate-700' : 'bg-slate-50 border border-slate-100 text-slate-400 group-hover:bg-white group-hover:shadow-sm'} group-hover:${theme.text} group-hover:${theme.border}`}>
                    <sub.icon size={18} />
                  </div>
                  <div className="ml-4 flex-1">
                    <span className={`block text-sm font-bold group-hover:${theme.text} transition-colors ${isDarkMode ? 'text-slate-200' : 'text-slate-700'}`}>{sub.label}</span>
                    <span className="block text-[10px] text-slate-500 font-medium uppercase tracking-tight">Access files</span>
                  </div>
                  <ChevronRight size={14} className={`text-slate-300 opacity-0 group-hover:opacity-100 transition-all transform ${isLeft ? 'group-hover:translate-x-1' : 'rotate-180 group-hover:-translate-x-1'} group-hover:${theme.text}`} />
                </button>
              ))
            )}
          </div>
          {panelItem !== 'classes-panel' && (
            <div className={`p-4 border-t ${isDarkMode ? 'bg-slate-900/50 border-slate-800' : 'bg-slate-50/50 border-slate-100'}`}>
              <button className={`w-full py-3 px-4 rounded-xl border-2 border-dashed text-xs font-bold transition-all flex items-center justify-center ${isDarkMode ? 'border-slate-800 text-slate-500 hover:border-slate-700 hover:text-slate-300' : 'border-slate-200 text-slate-400 hover:border-slate-300 hover:text-slate-600'} hover:${theme.border} hover:${theme.text}`}>
                <Plus size={14} className="mr-2" /> Create New
              </button>
            </div>
          )}
        </div>
      </div>

      {/* MAIN CONTENT AREA */}
      <div className="flex-1 flex flex-col min-w-0 relative z-10">
        <header className={`border-b flex items-center px-6 z-20 shrink-0 ${navHeight} sticky top-0 transition-colors duration-300 ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
          
          {/* LEFT: STATIC BRANDING SECTION */}
          <div 
            className={`transition-all duration-500 overflow-hidden flex items-center shrink-0 ${isSearchExpanded && !useOverlaySearch ? 'w-0 opacity-0 mr-0' : 'flex-1 opacity-100 mr-4'}`}
          >
            <div className="flex items-center text-lg font-bold whitespace-nowrap tracking-tight">
              <span className={isDarkMode ? 'text-white' : 'text-slate-900'}>Edu.</span>
              <span className={theme.text}>{activeApp}</span>
            </div>
          </div>
          
          {/* MOBILE OVERLAY SEARCH */}
          {useOverlaySearch && isSearchExpanded && (
            <div className={`absolute inset-x-0 inset-y-0 z-50 flex items-center px-4 animate-in fade-in slide-in-from-top-2 duration-300 ${isDarkMode ? 'bg-slate-900' : 'bg-white'}`}>
                <Search size={18} className="text-slate-400 mr-3" />
                <input
                    type="text"
                    placeholder="Search..."
                    autoFocus
                    className={`flex-1 bg-transparent border-none outline-none text-sm font-medium ${isDarkMode ? 'text-white' : 'text-slate-900'}`}
                    value={searchQuery}
                    onChange={(e) => setSearchQuery(e.target.value)}
                />
                <button onClick={() => setIsSearchExpanded(false)} className="p-2 text-slate-400 hover:text-slate-900">
                    <X size={18} />
                </button>
            </div>
          )}
          
          {/* NAV ACTIONS */}
          <div className={`flex items-center justify-end flex-1 h-full relative`}>
             
             {!useOverlaySearch && (
               <div 
                  ref={searchRef} 
                  className={`flex items-center relative transition-all duration-500 ease-in-out h-10 justify-end
                    ${isSearchExpanded ? 'flex-1 max-w-xl mr-4' : 'w-10'}`}
               >
                  <div className={`relative h-full transition-all duration-500 overflow-hidden flex items-center ${isSearchExpanded ? 'w-full' : 'w-0'}`}>
                    <div className={`absolute left-0 flex items-center justify-center w-12 h-10 z-30 ${isSearchExpanded ? theme.text : 'text-slate-400'}`}>
                        <Search size={18} />
                    </div>

                    <input
                      type="text"
                      placeholder="Search resources..."
                      value={searchQuery}
                      onChange={(e) => setSearchQuery(e.target.value)}
                      className={`w-full h-full border rounded-full pl-12 pr-12 text-sm font-medium outline-none focus:outline-none transition-all duration-300 ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white focus:bg-slate-700 focus:border-slate-600' : 'bg-slate-50 border-slate-200 text-slate-900 focus:bg-white focus:border-slate-300'}`}
                      autoFocus={isSearchExpanded}
                    />
                    
                    <div className="absolute right-0 flex items-center justify-center w-12 h-10 z-30">
                        <button 
                            onClick={(e) => { 
                                e.stopPropagation();
                                if (searchQuery) {
                                    setSearchQuery("");
                                } else {
                                    setIsSearchExpanded(false);
                                }
                            }}
                            className={`p-1.5 text-slate-400 hover:text-slate-900 transition-all duration-300 transform
                            ${isSearchExpanded ? 'opacity-100 scale-100' : 'opacity-0 scale-50 pointer-events-none'}`}
                        >
                            <X size={16} strokeWidth={2.5} />
                        </button>
                    </div>
                  </div>

                  <button 
                    onClick={() => setIsSearchExpanded(true)}
                    className={`absolute right-0 p-2 text-slate-400 hover:${theme.text} rounded-full transition-all duration-300 z-20 
                      ${isSearchExpanded ? 'opacity-0 pointer-events-none' : 'opacity-100'}`}
                  >
                    <Search size={18} />
                  </button>
               </div>
             )}

             {useOverlaySearch && !isSearchExpanded && (
                <button onClick={() => setIsSearchExpanded(true)} className="p-2 text-slate-400 hover:text-slate-900 rounded-full mr-2">
                    <Search size={18} />
                </button>

              <div className="flex items-center">
                {modalStep === 2 && (
                   <button onClick={() => setModalStep(1)} className={`mr-2 p-1 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:bg-slate-800' : 'text-slate-500 hover:bg-slate-100'}`}>
                      <ArrowLeft size={16} strokeWidth={3} />
                   </button>
                )}
                <h3 className={`font-bold text-lg ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>
                  {modalStep === 1 ? 'Create New Class' : 'Add Students'}
                </h3>
              </div>
              <button 
                onClick={() => {
                  setIsAddClassModalOpen(false);
                  setModalStep(1);
                  setStudentAddMethod(null);
                  setManualStudents('');
                  setCsvFileName('');
                  setParsedCsvStudents([]);
                }} 
                className={`p-1.5 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:text-white hover:bg-slate-800' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}
              >
                <X size={20} />
              </button>
            </div>
            
            {modalStep === 1 ? (
              <div className="p-6 space-y-4">
                <div>
                  <label className={`block text-sm font-semibold mb-2 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Class Icon</label>
                  <div className="flex flex-wrap gap-2">
                    {Object.keys(iconMap).map(iconName => {
                      const IconComp = iconMap[iconName];
                      const isSelected = newClassForm.icon === iconName;
                      return (
                        <button
                          key={iconName}
                          onClick={() => setNewClassForm({...newClassForm, icon: iconName})}
                          className={`w-10 h-10 rounded-xl flex items-center justify-center transition-all ${isSelected ? `${theme.bg} text-white shadow-md` : isDarkMode ? 'bg-slate-800 text-slate-400 hover:bg-slate-700 hover:text-slate-300' : 'bg-slate-100 text-slate-500 hover:bg-slate-200 hover:text-slate-700'}`}
                          type="button"
                        >
                          <IconComp size={20} strokeWidth={isSelected ? 2.5 : 2} />
                        </button>
                      )
                    })}
                  </div>
                </div>
                <div>
                  <label className={`block text-sm font-semibold mb-1.5 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Class Name</label>
                  <input 
                    type="text" 
                    placeholder="e.g. Advanced Biology"
                    value={newClassForm.name}
                    onChange={e => setNewClassForm({...newClassForm, name: e.target.value})}
                    className={`w-full px-4 py-2.5 rounded-xl border text-sm transition-all outline-none focus:ring-2 focus:ring-opacity-50 focus:border-transparent ${theme.ring} ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white placeholder-slate-500' : 'bg-slate-50 border-slate-200 text-slate-900 placeholder-slate-400'}`}
                  />
                </div>
                <div>
                  <label className={`block text-sm font-semibold mb-1.5 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Subject</label>
                  <input 
                    type="text" 
                    placeholder="e.g. Science"
                    value={newClassForm.subject}
                    onChange={e => setNewClassForm({...newClassForm, subject: e.target.value})}
                    className={`w-full px-4 py-2.5 rounded-xl border text-sm transition-all outline-none focus:ring-2 focus:ring-opacity-50 focus:border-transparent ${theme.ring} ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white placeholder-slate-500' : 'bg-slate-50 border-slate-200 text-slate-900 placeholder-slate-400'}`}
                  />
                </div>
                <div>
                  <label className={`block text-sm font-semibold mb-1.5 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Grade Level</label>
                  <select 
                    value={newClassForm.grade}
                    onChange={e => setNewClassForm({...newClassForm, grade: e.target.value})}
                    className={`w-full px-4 py-2.5 rounded-xl border text-sm transition-all outline-none focus:ring-2 focus:ring-opacity-50 focus:border-transparent appearance-none ${theme.ring} ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white' : 'bg-slate-50 border-slate-200 text-slate-900'}`}
                  >
                    <option value="" disabled>Select a grade level...</option>
                    <option value="Pre-K">Pre-K</option>
                    <option value="Kindergarten">Kindergarten</option>
                    <option value="1st Grade">1st Grade</option>
                    <option value="2nd Grade">2nd Grade</option>
                    <option value="3rd Grade">3rd Grade</option>
                    <option value="4th Grade">4th Grade</option>
                    <option value="5th Grade">5th Grade</option>
                    <option value="6th Grade">6th Grade</option>
                    <option value="7th Grade">7th Grade</option>
                    <option value="8th Grade">8th Grade</option>
                    <option value="9th Grade">9th Grade</option>
                    <option value="10th Grade">10th Grade</option>
                    <option value="11th Grade">11th Grade</option>
                    <option value="12th Grade">12th Grade</option>
                    <option value="Other">Other</option>
                  </select>
                </div>
              </div>
            ) : (
              <div className="p-6 space-y-4 animate-in slide-in-from-right-4 duration-300">
                <p className={`text-sm ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>How would you like to import your class roster?</p>
                
                <div className="space-y-3">
                  <button 
                    onClick={() => setStudentAddMethod('csv')} 
                    className={`w-full p-4 border rounded-xl flex items-center transition-all ${studentAddMethod === 'csv' ? `${theme.border} ${isDarkMode ? 'bg-slate-800/80' : 'bg-slate-50'} ring-1 ${theme.ring}` : `${isDarkMode ? 'border-slate-800 hover:border-slate-700 hover:bg-slate-800/40' : 'border-slate-200 hover:border-slate-300 hover:bg-slate-50/50'}`}`}
                  >
                    <div className={`p-2 rounded-lg mr-4 ${studentAddMethod === 'csv' ? `${theme.bg} text-white` : isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-100 text-slate-500'}`}>
                       <UploadCloud size={20} />
                    </div>
                    <div className="text-left flex-1">
                      <p className={`font-bold text-sm ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Upload CSV</p>
                      <p className={`text-xs ${isDarkMode ? 'text-slate-500' : 'text-slate-500'}`}>Import from a spreadsheet file</p>
                    </div>
                  </button>

                  <button 
                    onClick={() => setStudentAddMethod('google')} 
                    className={`w-full p-4 border rounded-xl flex items-center transition-all ${studentAddMethod === 'google' ? `${theme.border} ${isDarkMode ? 'bg-slate-800/80' : 'bg-slate-50'} ring-1 ${theme.ring}` : `${isDarkMode ? 'border-slate-800 hover:border-slate-700 hover:bg-slate-800/40' : 'border-slate-200 hover:border-slate-300 hover:bg-slate-50/50'}`}`}
                  >
                    <div className={`p-2 rounded-lg mr-4 ${studentAddMethod === 'google' ? `${theme.bg} text-white` : isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-100 text-slate-500'}`}>
                       <Link2 size={20} />
                    </div>
                    <div className="text-left flex-1">
                      <p className={`font-bold text-sm ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Google Classroom</p>
                      <p className={`text-xs ${isDarkMode ? 'text-slate-500' : 'text-slate-500'}`}>Sync with an existing classroom</p>
                    </div>
                  </button>

                  <button 
                    onClick={() => setStudentAddMethod('manual')} 
                    className={`w-full p-4 border rounded-xl flex items-center transition-all ${studentAddMethod === 'manual' ? `${theme.border} ${isDarkMode ? 'bg-slate-800/80' : 'bg-slate-50'} ring-1 ${theme.ring}` : `${isDarkMode ? 'border-slate-800 hover:border-slate-700 hover:bg-slate-800/40' : 'border-slate-200 hover:border-slate-300 hover:bg-slate-50/50'}`}`}
                  >
                    <div className={`p-2 rounded-lg mr-4 ${studentAddMethod === 'manual' ? `${theme.bg} text-white` : isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-100 text-slate-500'}`}>

                         if (e.target.files && e.target.files[0]) {
                            const file = e.target.files[0];
                            setCsvFileName(file.name);
                            
                            // Read and parse the CSV file
                            const reader = new FileReader();
                            reader.onload = (event) => {
                               const text = event.target.result;
                               const lines = text.split(/\r?\n/).filter(line => line.trim() !== '');
                               if (lines.length === 0) return;

                               // Helper to properly split CSV rows ignoring commas inside quotes
                               const parseCSVLine = (line) => {
                                 const result = [];
                                 let current = '';
                                 let inQuotes = false;
                                 for (let i = 0; i < line.length; i++) {
                                   const char = line[i];
                                   if (char === '"') inQuotes = !inQuotes;
                                   else if (char === ',' && !inQuotes) { result.push(current.trim()); current = ''; }
                                   else current += char;
                                 }
                                 result.push(current.trim());
                                 return result.map(s => s.replace(/(^"|"$)/g, '')); // Strip surrounding quotes
                               };

                               const headers = parseCSVLine(lines[0].toLowerCase());
                               let nameColIdx = headers.findIndex(h => h === 'name' || h === 'student name' || h === 'student');
                               let firstColIdx = headers.findIndex(h => h.includes('first'));
                               let lastColIdx = headers.findIndex(h => h.includes('last'));

                               // If we found header matches, we skip the first row (index 0). Otherwise, parse from row 0.
                               const startIndex = (nameColIdx !== -1 || firstColIdx !== -1 || lastColIdx !== -1 || headers.some(h => h.includes('id') || h.includes('course'))) ? 1 : 0;
                               
                               const parsedNames = [];

                               for (let i = startIndex; i < lines.length; i++) {
                                  const cols = parseCSVLine(lines[i]);
                                  let studentName = '';
                                  
                                  // 1. Try to pull by known header columns
                                  if (firstColIdx !== -1 && lastColIdx !== -1) {
                                     studentName = `${cols[firstColIdx]} ${cols[lastColIdx]}`;
                                  } else if (nameColIdx !== -1) {
                                     studentName = cols[nameColIdx];
                                  } else {
                                     // 2. Fallback: Scan row for something that looks exactly like a name (e.g. "Harper Anderson")
                                     for (const col of cols) {
                                        const cleanCol = col.replace(/['"-]/g, '').trim();
                                        // Looks for 2 or 3 words, alphabet only, no numbers
                                        if (/^[a-zA-Z]+ [a-zA-Z]+( [a-zA-Z]+)?$/.test(cleanCol) && cleanCol.length < 30) {
                                           studentName = col;
                                           break;
                                        }
                                     }
                                  }
                                  
                                  if (studentName && studentName.trim() && !studentName.toLowerCase().includes('undefined')) {
                                     // Build object with all properties
                                     let studentData = { name: studentName.trim() };
                                     
                                     if (startIndex === 1) {
                                        headers.forEach((h, idx) => {
                                           if (idx !== nameColIdx && idx !== firstColIdx && idx !== lastColIdx && cols[idx]) {
                                              const cleanKey = h.trim().replace(/[^a-z0-9]/gi, '_').toLowerCase();
                                              if (cleanKey) studentData[cleanKey] = cols[idx].trim();
                                           }
                                        });
                                     } else {
                                        cols.forEach((col, idx) => {
                                           if (col.trim() !== studentData.name && col.trim() !== '') {
                                              studentData[`col_${idx}`] = col.trim();
                                           }
                                        });
                                     }

                                     parsedNames.push(studentData);
                                  }
                               }
                               
                               setParsedCsvStudents(parsedNames);
                            };
                            reader.readAsText(file);
                         }
                      }} 
                    />
                    <UploadCloud size={24} className={`mb-3 transition-colors ${csvFileName ? theme.text : 'opacity-50'}`} />
                    <p className="text-sm font-semibold mb-3">Drag and drop file here</p>
                    
                    {csvFileName ? (
                       <div className={`px-4 py-2 rounded-lg text-xs font-semibold border flex items-center ${isDarkMode ? 'bg-slate-900 border-slate-700' : 'bg-white border-slate-200'}`}>
                         <span className="truncate max-w-[150px]">{csvFileName}</span>
                         <button onClick={(e) => { e.stopPropagation(); setCsvFileName(''); setParsedCsvStudents([]); fileInputRef.current.value = ''; }} className="ml-2 hover:text-rose-500 transition-colors"><X size={14}/></button>
                       </div>
                    ) : (
                       <button 
                         onClick={() => fileInputRef.current?.click()}
                         className={`px-4 py-2 rounded-lg text-xs font-bold transition-all shadow-sm text-white hover:shadow-md hover:-translate-y-0.5 ${theme.bg}`}
                       >
                         Browse Files
                       </button>
                    )}
                    <p className="text-[10px] mt-3 opacity-60">Supports .csv files only</p>
                  </div>
                )}
                {studentAddMethod === 'google' && (
                  <div className={`mt-4 p-4 border rounded-xl flex items-center justify-between animate-in fade-in ${isDarkMode ? 'border-slate-800 bg-slate-800/50' : 'border-slate-200 bg-slate-50/50'}`}>
                     <div className="flex items-center">
                        <div className="w-8 h-8 rounded bg-emerald-100 text-emerald-600 flex items-center justify-center mr-3"><BookOpen size={16}/></div>
                        <span className={`text-sm font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Connect Google Account</span>
                     </div>
                     <button className={`px-4 py-2 rounded-lg text-xs font-bold text-white shadow-sm transition-all hover:-translate-y-0.5 ${theme.bg}`}>Connect</button>
                  </div>
                )}
                {studentAddMethod === 'manual' && (
                  <div className="mt-4 animate-in fade-in">
                    <label className={`block text-xs font-semibold mb-1.5 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Student Names (Comma or line separated)</label>
                    <textarea 
                      rows="4" 
                      placeholder={"e.g.\nEleanor Shellstrop, Chidi Anagonye\nJason Mendoza"}
                      value={manualStudents}
                      onChange={(e) => setManualStudents(e.target.value)}
                      className={`w-full px-4 py-3 rounded-xl border text-sm transition-all outline-none focus:ring-2 focus:ring-opacity-50 focus:border-transparent resize-none ${theme.ring} ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white placeholder-slate-600' : 'bg-slate-50 border-slate-200 text-slate-900 placeholder-slate-400'}`}
                    ></textarea>
                  </div>
                )}
              </div>
            )}
            <div className={`px-6 py-4 border-t flex items-center justify-end space-x-3 ${isDarkMode ? 'border-slate-800 bg-slate-900/50' : 'border-slate-100 bg-slate-50/50'}`}>
              {modalStep === 1 ? (
                <>
                  <button 
                    onClick={() => {
                      setIsAddClassModalOpen(false);
                      setModalStep(1);
                      setStudentAddMethod(null);
                      setManualStudents('');
                      setCsvFileName('');
                      setParsedCsvStudents([]);
                    }}
                    className={`px-4 py-2 rounded-xl text-sm font-bold transition-colors ${isDarkMode ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-600 hover:bg-slate-200'}`}
                  >
                    Cancel
                  </button>
                  <button 
                    onClick={() => {
                      if (!newClassForm.name.trim()) return; 
                      setModalStep(2);
                    }}
                    className={`px-6 py-2 rounded-xl text-sm font-bold text-white shadow-sm transition-all hover:shadow-md hover:-translate-y-0.5 ${theme.bg}`}
                  >
                    Next Step
                  </button>
                </>
              ) : (
                <>
                  <button 
                    onClick={() => setModalStep(1)}
                    className={`px-4 py-2 rounded-xl text-sm font-bold transition-colors ${isDarkMode ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-600 hover:bg-slate-200'}`}
                  >
                    Back
                  </button>
                  <button 
                    onClick={() => {
                      let newStudentList = [];
                      if (studentAddMethod === 'manual' && manualStudents.trim()) {
                         newStudentList = manualStudents.split(/[\n,]/).map(s => s.trim()).filter(s => s !== '').map((name, i) => ({ id: `manual-${Date.now()}-${i}`, name }));
                      } else if (studentAddMethod === 'csv') {
                         newStudentList = parsedCsvStudents.map((s, i) => ({ ...s, id: s.id || s.student_id || s.id_number || `csv-${Date.now()}-${i}` }));
                      } else if (studentAddMethod === 'google') {
                         newStudentList = Array.from({length: 20}, (_, i) => ({ name: `Google Student ${i+1}`, id: `G${1000 + i}` })); // Mock count for demo purposes
                      }
                      
                      const newAppCodes = {
                          app1: Math.random().toString(36).substring(2, 8).toUpperCase(),
                          app2: Math.random().toString(36).substring(2, 8).toUpperCase(),
                          app3: Math.random().toString(36).substring(2, 8).toUpperCase(),
                          app4: Math.random().toString(36).substring(2, 8).toUpperCase(),
                      };
                      
                      setClasses([...classes, { id: Date.now(), ...newClassForm, isArchived: false, appCodes: newAppCodes, studentList: newStudentList }]);
                      
                      setIsAddClassModalOpen(false);
                      setModalStep(1);
                      setStudentAddMethod(null);
                      setManualStudents('');
                      setCsvFileName('');
                      setParsedCsvStudents([]);
                      setNewClassForm({ name: '', subject: '', grade: '', icon: 'BookOpen' });
                    }}
                    className={`px-6 py-2 rounded-xl text-sm font-bold text-white shadow-sm transition-all hover:shadow-md hover:-translate-y-0.5 ${theme.bg}`}
                  >
                    Finish & Create
                  </button>
                </>
              )}
            </div>
          </div>
        </div>
      )}

      {/* EMOJI PICKER MODAL */}
      {isEmojiPickerOpen && (
        <div className="fixed inset-0 z-[110] flex items-center justify-center p-4 animate-in fade-in duration-200">
           <div className="absolute inset-0 bg-slate-900/60 backdrop-blur-sm transition-opacity" onClick={() => setIsEmojiPickerOpen(false)}></div>
           <div className={`relative w-full max-w-xs rounded-2xl shadow-2xl overflow-hidden flex flex-col animate-in zoom-in-95 duration-200 ${isDarkMode ? 'bg-slate-900 border border-slate-800' : 'bg-white'}`}>
              <div className={`px-4 py-3 border-b flex justify-between items-center ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
                 <h3 className={`font-bold text-sm ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Choose Avatar</h3>
                 <button onClick={() => setIsEmojiPickerOpen(false)} className={`p-1.5 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:text-white hover:bg-slate-800' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}>
                    <X size={16} />
                 </button>
              </div>
              <div className="p-4 grid grid-cols-6 gap-2 max-h-64 overflow-y-auto">
                 {EMOJI_LIST.map((emoji, index) => (
                    <button
                       key={`${emoji}-${index}`}
                       onClick={() => handleEmojiSelect(emoji)}
                       className={`w-10 h-10 flex items-center justify-center text-2xl rounded-xl transition-all hover:scale-125 hover:z-10 ${isDarkMode ? 'hover:bg-slate-800' : 'hover:bg-slate-50'}`}
                    >
                       {emoji}
                    </button>
                 ))}
              </div>
           </div>
        </div>
      )}

      {/* GENERAL SETTINGS MODAL */}
      {isSettingsModalOpen && (
        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 animate-in fade-in duration-200">
          <div className="absolute inset-0 bg-slate-900/60 backdrop-blur-sm transition-opacity" onClick={() => setIsSettingsModalOpen(false)}></div>
          <div className={`relative w-full max-w-md rounded-2xl shadow-2xl overflow-hidden flex flex-col animate-in zoom-in-95 duration-200 ${isDarkMode ? 'bg-slate-900 border border-slate-800' : 'bg-white'}`}>
            <div className={`px-6 py-4 border-b flex justify-between items-center ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
              <h3 className={`font-bold text-lg ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>General Settings</h3>
              <button onClick={() => setIsSettingsModalOpen(false)} className={`p-1.5 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:text-white hover:bg-slate-800' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}>
                <X size={20} />
              </button>
            </div>
            
            <div className="p-6 space-y-4">
              <div>
                <label className={`block text-sm font-semibold mb-2 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Class Icon</label>
                <div className="flex flex-wrap gap-2">
                  {Object.keys(iconMap).map(iconName => {
                    const IconComp = iconMap[iconName];
                    const isSelected = editClassForm.icon === iconName;
                    return (
                      <button
                        key={iconName}
                        onClick={() => setEditClassForm({...editClassForm, icon: iconName})}
                        className={`w-10 h-10 rounded-xl flex items-center justify-center transition-all ${isSelected ? `${theme.bg} text-white shadow-md` : isDarkMode ? 'bg-slate-800 text-slate-400 hover:bg-slate-700 hover:text-slate-300' : 'bg-slate-100 text-slate-500 hover:bg-slate-200 hover:text-slate-700'}`}
                        type="button"
                      >
                        <IconComp size={20} strokeWidth={isSelected ? 2.5 : 2} />
                      </button>
                    )
                  })}
                </div>
              </div>
              <div>
                <label className={`block text-sm font-semibold mb-1.5 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Class Name</label>
                <input 
                  type="text" 
                  value={editClassForm.name}
                  onChange={e => setEditClassForm({...editClassForm, name: e.target.value})}
                  className={`w-full px-4 py-2.5 rounded-xl border text-sm transition-all outline-none focus:ring-2 focus:ring-opacity-50 focus:border-transparent ${theme.ring} ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white placeholder-slate-500' : 'bg-slate-50 border-slate-200 text-slate-900 placeholder-slate-400'}`}
                />
              </div>
              <div>
                <label className={`block text-sm font-semibold mb-1.5 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Subject</label>
                <input 
                  type="text" 
                  value={editClassForm.subject}
                  onChange={e => setEditClassForm({...editClassForm, subject: e.target.value})}
                  className={`w-full px-4 py-2.5 rounded-xl border text-sm transition-all outline-none focus:ring-2 focus:ring-opacity-50 focus:border-transparent ${theme.ring} ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white placeholder-slate-500' : 'bg-slate-50 border-slate-200 text-slate-900 placeholder-slate-400'}`}
                />
              </div>
              <div>
                <label className={`block text-sm font-semibold mb-1.5 ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Grade Level</label>
                <select 
                  value={editClassForm.grade}
                  onChange={e => setEditClassForm({...editClassForm, grade: e.target.value})}
                  className={`w-full px-4 py-2.5 rounded-xl border text-sm transition-all outline-none focus:ring-2 focus:ring-opacity-50 focus:border-transparent appearance-none ${theme.ring} ${isDarkMode ? 'bg-slate-800 border-slate-700 text-white' : 'bg-slate-50 border-slate-200 text-slate-900'}`}
                >
                  <option value="" disabled>Select a grade level...</option>
                  <option value="Pre-K">Pre-K</option>
                  <option value="Kindergarten">Kindergarten</option>
                  <option value="1st Grade">1st Grade</option>
                  <option value="2nd Grade">2nd Grade</option>
                  <option value="3rd Grade">3rd Grade</option>
                  <option value="4th Grade">4th Grade</option>
                  <option value="5th Grade">5th Grade</option>
                  <option value="6th Grade">6th Grade</option>
                  <option value="7th Grade">7th Grade</option>
                  <option value="8th Grade">8th Grade</option>
                  <option value="9th Grade">9th Grade</option>
                  <option value="10th Grade">10th Grade</option>
                  <option value="11th Grade">11th Grade</option>
                  <option value="12th Grade">12th Grade</option>
                  <option value="Other">Other</option>
                </select>
              </div>
            </div>
            
            <div className={`px-6 py-4 border-t flex items-center justify-end space-x-3 ${isDarkMode ? 'border-slate-800 bg-slate-900/50' : 'border-slate-100 bg-slate-50/50'}`}>
              <button 
                onClick={() => setIsSettingsModalOpen(false)}
                className={`px-4 py-2 rounded-xl text-sm font-bold transition-colors ${isDarkMode ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-600 hover:bg-slate-200'}`}
              >
                Cancel
              </button>
              <button 
                onClick={() => {
                  if (!editClassForm.name.trim()) return;
                  const updatedClass = { ...selectedClass, ...editClassForm };
                  setSelectedClass(updatedClass);
                  setClasses(classes.map(c => c.id === updatedClass.id ? updatedClass : c));
                  setIsSettingsModalOpen(false);
                }}
                className={`px-6 py-2 rounded-xl text-sm font-bold text-white shadow-sm transition-all hover:shadow-md hover:-translate-y-0.5 ${theme.bg}`}
              >
                Save Changes
              </button>
            </div>
          </div>
        </div>
      )}

      {/* IN-DEPTH APP SETTINGS MODAL */}
      {isAppSettingModalOpen && selectedAppSetting && (
        <div className="fixed inset-0 z-[100] flex items-center justify-center p-4 animate-in fade-in duration-200">
          <div className="absolute inset-0 bg-slate-900/60 backdrop-blur-sm transition-opacity" onClick={() => setIsAppSettingModalOpen(false)}></div>
          <div className={`relative w-full max-w-lg rounded-2xl shadow-2xl overflow-hidden flex flex-col animate-in zoom-in-95 duration-200 ${isDarkMode ? 'bg-slate-900 border border-slate-800' : 'bg-white'}`}>
            <div className={`px-6 py-4 border-b flex justify-between items-center ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
              <div className="flex items-center">
                  <div className={`w-8 h-8 rounded-lg flex items-center justify-center ${selectedAppSetting.bg} bg-opacity-10 mr-3`}>
                      <selectedAppSetting.icon size={16} className={selectedAppSetting.color} />
                  </div>
                  <h3 className={`font-bold text-lg ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>{selectedAppSetting.name} Advanced Settings</h3>
              </div>
              <button onClick={() => setIsAppSettingModalOpen(false)} className={`p-1.5 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:text-white hover:bg-slate-800' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}>
                <X size={20} />
              </button>
            </div>
            
            <div className="p-6 space-y-6">
               <div>
                  <h4 className={`text-sm font-bold mb-3 ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Access Controls</h4>
                  <div className="space-y-3">
                      <label className={`flex items-center justify-between p-3.5 rounded-xl border cursor-pointer transition-colors ${isDarkMode ? 'border-slate-800 bg-slate-800/30 hover:bg-slate-800/50' : 'border-slate-200 bg-slate-50 hover:bg-slate-100'}`}>
                          <span className={`text-sm font-semibold ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Allow off-hours access</span>
                          <input type="checkbox" className={`rounded border-slate-300 focus:ring-2 focus:ring-opacity-50 cursor-pointer w-4 h-4 ${theme.text} ${theme.ring}`} />
                      </label>
                      <label className={`flex items-center justify-between p-3.5 rounded-xl border cursor-pointer transition-colors ${isDarkMode ? 'border-slate-800 bg-slate-800/30 hover:bg-slate-800/50' : 'border-slate-200 bg-slate-50 hover:bg-slate-100'}`}>
                          <span className={`text-sm font-semibold ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Require School Network IP</span>
                          <input type="checkbox" defaultChecked className={`rounded border-slate-300 focus:ring-2 focus:ring-opacity-50 cursor-pointer w-4 h-4 ${theme.text} ${theme.ring}`} />
                      </label>
                  </div>
               </div>
               
               <div>
                  <h4 className={`text-sm font-bold mb-3 ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Data & Syncing</h4>
                  <div className="space-y-3">
                      <label className={`flex items-center justify-between p-3.5 rounded-xl border cursor-pointer transition-colors ${isDarkMode ? 'border-slate-800 bg-slate-800/30 hover:bg-slate-800/50' : 'border-slate-200 bg-slate-50 hover:bg-slate-100'}`}>
                          <span className={`text-sm font-semibold ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Sync grades automatically</span>
                          <input type="checkbox" defaultChecked className={`rounded border-slate-300 focus:ring-2 focus:ring-opacity-50 cursor-pointer w-4 h-4 ${theme.text} ${theme.ring}`} />
                      </label>
                  </div>
               </div>
            </div>
            
            <div className={`px-6 py-4 border-t flex items-center justify-end space-x-3 ${isDarkMode ? 'border-slate-800 bg-slate-900/50' : 'border-slate-100 bg-slate-50/50'}`}>
              <button 
                onClick={() => setIsAppSettingModalOpen(false)}
                className={`px-4 py-2 rounded-xl text-sm font-bold transition-colors ${isDarkMode ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-600 hover:bg-slate-200'}`}
              >
                Close
              </button>
              <button 
                onClick={() => setIsAppSettingModalOpen(false)}
                className={`px-6 py-2 rounded-xl text-sm font-bold text-white shadow-sm transition-all hover:shadow-md hover:-translate-y-0.5 ${theme.bg}`}
              >
                Save Settings
              </button>
            </div>
          </div>
        </div>
      )}

      </div>
    </ClassContext.Provider>
  );
};

export default App;
