/**
 * Holiday Helper
 */
const getHoliday = (date) => {
  if (!date) return null;
  const month = date.getMonth();
  const day = date.getDate();
  const dayOfWeek = date.getDay();

  if (month === 0 && day === 1) return "New Year's Day";
  if (month === 1 && day === 14) return "Valentine's Day";
  if (month === 2 && day === 17) return "St. Patrick's Day";
  if (month === 5 && day === 19) return "Juneteenth";
  if (month === 6 && day === 4) return "Independence Day";
  if (month === 9 && day === 31) return "Halloween";
  if (month === 10 && day === 11) return "Veterans Day";
  if (month === 11 && day === 25) return "Christmas";
  if (month === 11 && day === 31) return "New Year's Eve";

  const isNthDayOfWeek = (n, targetDayOfWeek) => {
    return dayOfWeek === targetDayOfWeek && Math.ceil(day / 7) === n;
  };
  const isLastMonday = () => {
    if (dayOfWeek !== 1) return false;
    const daysInMonth = new Date(date.getFullYear(), month + 1, 0).getDate();
    return day + 7 > daysInMonth;
  };

  if (month === 0 && isNthDayOfWeek(3, 1)) return "MLK Jr. Day";
  if (month === 1 && isNthDayOfWeek(3, 1)) return "Presidents' Day";
  if (month === 4 && isLastMonday()) return "Memorial Day";
  if (month === 8 && isNthDayOfWeek(1, 1)) return "Labor Day";
  if (month === 10 && isNthDayOfWeek(4, 4)) return "Thanksgiving";

  return null;
};

/**
 * Time Helper for Duration Calculation
 */
const parseTime = (timeStr, ampmStr) => {
  if (!timeStr) return 0;
  let [h, m] = timeStr.split(':').map(Number);
  if (isNaN(h)) h = 0;
  if (isNaN(m)) m = 0;
  if (ampmStr === 'PM' && h !== 12) h += 12;
  if (ampmStr === 'AM' && h === 12) h = 0;
  return h + (m / 60);
};

const getFormattedDate = (dateObj) => {
  if (!dateObj) return "";
  return `${dateObj.getFullYear()}-${String(dateObj.getMonth() + 1).padStart(2, '0')}-${String(dateObj.getDate()).padStart(2, '0')}`;
};

/**
 * Recurrence Engine Helpers
 */
const parseLocalDate = (dateStr) => {
  if (!dateStr) return new Date();
  const [y, m, d] = dateStr.split('-').map(Number);
  return new Date(y, m - 1, d);
};

const getEventsForDate = (dateStr, allEvents) => {
  const targetDate = parseLocalDate(dateStr);
  const targetTime = targetDate.getTime();
  
  return allEvents.filter(e => {
    if (e.excludedDates && e.excludedDates.includes(dateStr)) return false;

    const eventStart = parseLocalDate(e.date);
    const eventEnd = e.endDate ? parseLocalDate(e.endDate) : eventStart;
    const durationDays = Math.round((eventEnd.getTime() - eventStart.getTime()) / 86400000);
    
    if (!e.isRecurring) {
      return targetTime >= eventStart.getTime() && targetTime <= eventEnd.getTime();
    }
    
    if (targetTime < eventStart.getTime()) return false;
    
    const diffDays = Math.round((targetTime - eventStart.getTime()) / 86400000);
    
    if (e.frequency === 'Weekly') {
      const rem = diffDays % 7;
      return rem >= 0 && rem <= durationDays;
    } else if (e.frequency === 'Bi-Weekly') {
      const rem = diffDays % 14;
      return rem >= 0 && rem <= durationDays;
    } else if (e.frequency === 'Monthly') {
      let occStart = new Date(targetDate.getFullYear(), targetDate.getMonth(), eventStart.getDate());
      if (occStart.getTime() > targetTime) {
         occStart = new Date(targetDate.getFullYear(), targetDate.getMonth() - 1, eventStart.getDate());
      }
      const occEnd = new Date(occStart.getTime() + (durationDays * 86400000));
      return targetTime >= occStart.getTime() && targetTime <= occEnd.getTime();
    }
    
    return targetTime >= eventStart.getTime() && targetTime <= eventEnd.getTime();
  }).map(e => {
    const eventStart = parseLocalDate(e.date);
    const eventEnd = e.endDate ? parseLocalDate(e.endDate) : eventStart;
    const durationDays = Math.round((eventEnd.getTime() - eventStart.getTime()) / 86400000);
    
    let dayOfOccurrence = 0;
    if (!e.isRecurring) {
        dayOfOccurrence = Math.round((targetTime - eventStart.getTime()) / 86400000);
    } else {
        const diffDays = Math.round((targetTime - eventStart.getTime()) / 86400000);
        if (e.frequency === 'Weekly') dayOfOccurrence = diffDays % 7;
        else if (e.frequency === 'Bi-Weekly') dayOfOccurrence = diffDays % 14;
        else if (e.frequency === 'Monthly') {
            let occStart = new Date(targetDate.getFullYear(), targetDate.getMonth(), eventStart.getDate());
            if (occStart.getTime() > targetTime) {
                occStart = new Date(targetDate.getFullYear(), targetDate.getMonth() - 1, eventStart.getDate());
            }
            dayOfOccurrence = Math.round((targetTime - occStart.getTime()) / 86400000);
        }
    }
    
    return { ...e, occurrenceDate: dateStr, dayOfOccurrence, durationDays };
  }); 
};

             let end = d2;
             if (d1 > d2) { start = d2; end = d1; }

             newParsedBreaks.push({
               id: Date.now() + Math.random(), // Random factor for rapid batching
               name: name,
               startDate: getFormattedDate(start),
               endDate: getFormattedDate(end)
             });
          }
        } catch(e) { console.error('Failed to parse line:', line, e) }
      }

        breaks: [...prev.breaks, ...newParsedBreaks].sort((a, b) => new Date(a.startDate) - new Date(b.startDate))

'opacity-50' : ''}`}

'w-6 h-6 text-[10px]' : 'w-8 h-8 text-sm'} flex items-center justify-center rounded-full font-semibold transition-all ${isToday ? `${theme.bg} text-white shadow-md z-10` : (isDarkMode ?                        <Calendar size={22} />
                     </div>
                  </div>

                  <div className="grid grid-cols-1 md:grid-cols-2 gap-6 relative z-10">
                     <div>
                        <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>First Day of School</label>

'bg-slate-800 text-slate-400 hover:bg-slate-700 hover:text-white' : 'bg-slate-100 text-slate-500 hover:bg-slate-200 hover:text-slate-900')}`}

                               <path fill="#34A853" d="M12 23c2.97 0 5.46-.98 7.28-2.66l-3.57-2.77c-.98.66-2.23 1.06-3.71 1.06-2.86 0-5.29-1.93-6.16-4.53H2.18v2.84C3.99 20.53 7.7 23 12 23z"/>
                               <path fill="#FBBC05" d="M5.84 14.09c-.22-.66-.35-1.36-.35-2.09s.13-1.43.35-2.09V7.07H2.18C1.43 8.55 1 10.22 1 12s.43 3.45 1.18 4.93l2.85-2.22.81-.62z"/>
                               <path fill="#EA4335" d="M12 5.38c1.62 0 3.06.56 4.21 1.64l3.15-3.15C17.45 2.09 14.97 1 12 1 7.7 1 3.99 3.47 2.18 7.07l3.66 2.84c.87-2.6 3.3-4.53 6.16-4.53z"/>
                             </svg>
                           </div>
                           <div>
                             <p className={`text-sm font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Google Calendar</p>
                             <p className={`text-xs font-medium ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Not connected</p>
                           </div>
                         </div>
                         <button className={`px-4 py-2 rounded-xl text-xs font-bold transition-all ${isDarkMode ? 'bg-slate-800 text-white hover:bg-slate-700' : 'bg-white border border-slate-200 text-slate-700 hover:bg-slate-50 shadow-sm'}`}>Connect</button>
                      </div>

                      {/* Outlook */}
                      <div className={`flex items-center justify-between p-4 rounded-2xl border transition-all ${isDarkMode ? 'border-slate-800 bg-slate-950/50' : 'border-slate-100 bg-slate-50/50'}`}>
                         <div className="flex items-center space-x-4">
                           <div className="w-10 h-10 rounded-full bg-[#0078D4] shadow-sm flex items-center justify-center shrink-0">
                             <Mail size={18} className="text-white" />
                           </div>
                           <div>
                             <p className={`text-sm font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Outlook</p>
                             <p className={`text-xs font-medium ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Not connected</p>

                             </svg>
                           </div>
                           <div>
                             <p className={`text-sm font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Apple Calendar</p>

                           </div>
                         </div>
                         <button className={`px-4 py-2 rounded-xl text-xs font-bold transition-all ${isDarkMode ? 'bg-slate-800 text-white hover:bg-slate-700' : 'bg-white border border-slate-200 text-slate-700 hover:bg-slate-50 shadow-sm'}`}>Connect</button>
                      </div>
                    </div>
                  </div>
                </div>
              </div>
            ) : (
              <>
                <div className="h-64 rounded-3xl bg-slate-200/50 dark:bg-slate-800/30 animate-pulse" />
                <div className="grid grid-cols-1 md:grid-cols-3 gap-6">

        {activeTab === 'Create Calendar' && (

'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`}/>

'border-slate-800 text-slate-400 hover:bg-slate-800 hover:text-white' : 'border-slate-200 text-slate-600 hover:bg-slate-50'}`}>Block Schedule</button>

            <div className={`relative w-full max-w-sm rounded-3xl shadow-2xl overflow-hidden flex flex-col transform transition-all scale-100 ${isDarkMode ? 'bg-slate-900 border border-slate-800' : 'bg-white border border-slate-100'}`}>
              <div className={`p-6 border-b ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
                <div className="flex items-center justify-between">
                  <h3 className={`text-lg font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Cancel Recurring Event</h3>
                  <button onClick={() => setShowDeletePrompt(false)} className={`p-2 rounded-full transition-colors ${isDarkMode ? 'hover:bg-slate-800 text-slate-400 hover:text-white' : 'hover:bg-slate-100 text-slate-500 hover:text-slate-900'}`}>
                    <X size={18} />
                  </button>
                </div>
                <p className={`text-sm mt-2 font-medium ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>This is a repeating event. Do you want to cancel just this occurrence or the entire series?</p>
              </div>
              <div className="p-4 space-y-3 mb-2">
                <button onClick={handleDeleteSingle} className={`w-full py-3 px-4 rounded-xl font-bold transition-colors text-sm text-left ${isDarkMode ? 'bg-slate-800 text-white hover:bg-slate-700' : 'bg-slate-100 text-slate-900 hover:bg-slate-200'}`}>
                  Cancel This Event Only
                </button>
                <button onClick={handleDeleteSeries} className={`w-full py-3 px-4 rounded-xl font-bold transition-colors text-sm text-left ${isDarkMode ? 'bg-rose-500/10 text-rose-500 hover:bg-rose-500/20' : 'bg-rose-50 text-rose-600 hover:bg-rose-100'}`}>
                  Cancel Entire Series
                </button>
              </div>
            </div>
          </div>
        )}

        {/* Manage Breaks Modal */}
        {showBreaksModal && (
          <div className="fixed inset-0 z-[140] flex items-center justify-center p-4">
            <div className="absolute inset-0 bg-slate-900/40 backdrop-blur-sm transition-opacity" onClick={() => setShowBreaksModal(false)}></div>
            <div className={`relative w-full max-w-lg rounded-3xl shadow-2xl overflow-hidden flex flex-col transform transition-all scale-100 ${isDarkMode ? 'bg-slate-900 border border-slate-800' : 'bg-white border border-slate-100'}`}>
              <div className={`p-6 border-b ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
                <div className="flex items-center justify-between mb-4">
                  <h3 className={`text-lg font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>Manage Breaks & Holidays</h3>
                  <button onClick={() => setShowBreaksModal(false)} className={`p-2 rounded-full transition-colors ${isDarkMode ? 'hover:bg-slate-800 text-slate-400 hover:text-white' : 'hover:bg-slate-100 text-slate-500 hover:text-slate-900'}`}>
                    <X size={18} />
                  </button>
                </div>
                
                {/* Tabs for Input Methods */}
                <div className="flex space-x-1">
                  {['Manual', 'Paste List', 'Select'].map((tab) => (
                    <button
                      key={tab}
                      onClick={() => setBreaksTab(tab)}
                      className={`px-4 py-2 text-xs font-bold rounded-lg transition-all ${breaksTab === tab ? `${theme.bg} text-white shadow-sm` : (isDarkMode ? 'text-slate-400 hover:bg-slate-800 hover:text-slate-200' : 'text-slate-500 hover:bg-slate-100 hover:text-slate-900')}`}
                    >
                      {tab}
                    </button>
                  ))}
                </div>
              </div>
              
              <div className="p-6 space-y-6 max-h-[60vh] overflow-y-auto">
                {breaksTab === 'Manual' && (
                  <div className={`p-5 rounded-2xl border ${isDarkMode ? 'bg-slate-950/50 border-slate-800' : 'bg-slate-50/50 border-slate-200'}`}>
                    <h4 className={`text-[10px] font-bold uppercase tracking-widest mb-4 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Add Manual Break</h4>
                    <div className="space-y-4">
                      <div>
                        <input 
                          type="text" 
                          value={newBreak.name} 
                          onChange={(e) => setNewBreak({...newBreak, name: e.target.value})} 
                          placeholder="Break Label (e.g. Winter Break)" 
                          className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-900 border-slate-700 text-white focus:border-indigo-500' : 'bg-white border-slate-200 text-slate-900 focus:border-indigo-500'}`} 
                        />
                      </div>
                      <div className="grid grid-cols-2 gap-4">
                        <div>
                          <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Start Date</label>
                          <input 
                            type="date" 
                            value={newBreak.startDate} 
                            onChange={(e) => setNewBreak({...newBreak, startDate: e.target.value})} 
                            className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-900 border-slate-700 text-white focus:border-indigo-500' : 'bg-white border-slate-200 text-slate-900 focus:border-indigo-500'}`} 
                          />
                        </div>
                        <div>
                          <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>End Date</label>
                          <input 
                            type="date" 
                            value={newBreak.endDate} 
                            min={newBreak.startDate}
                            onChange={(e) => setNewBreak({...newBreak, endDate: e.target.value})} 
                            className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-900 border-slate-700 text-white focus:border-indigo-500' : 'bg-white border-slate-200 text-slate-900 focus:border-indigo-500'}`} 
                          />
                        </div>
                      </div>
                      <button 
                        onClick={handleAddBreak}
                        disabled={!newBreak.name || !newBreak.startDate || !newBreak.endDate}
                        className={`w-full py-3 rounded-xl font-bold transition-all text-sm flex items-center justify-center space-x-2 ${(!newBreak.name || !newBreak.startDate || !newBreak.endDate) ? (isDarkMode ? 'bg-slate-800 text-slate-500 cursor-not-allowed' : 'bg-slate-200 text-slate-400 cursor-not-allowed') : `${theme.bg} text-white shadow-md hover:opacity-90`}`}
                      >
                        <Plus size={16} />
                        <span>Add Break</span>
                      </button>
                    </div>
                  </div>
                )}

                {breaksTab === 'Paste List' && (
                  <div className={`p-5 rounded-2xl border ${isDarkMode ? 'bg-slate-950/50 border-slate-800' : 'bg-slate-50/50 border-slate-200'}`}>
                    <div className="flex items-center justify-between mb-4">
                      <h4 className={`text-[10px] font-bold uppercase tracking-widest ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Smart Paste</h4>
                      <span className={`text-[9px] font-medium ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>1 break per line</span>
                    </div>

                      <textarea

                        onChange={(e) => setPasteText(e.target.value)}
                        placeholder="e.g. Winter Break: 12/20/2024 to 01/02/2025&#10;Thanksgiving Nov 28, 2024"
                        className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold min-h-[120px] resize-none ${isDarkMode ? 'bg-slate-900 border-slate-700 text-white focus:border-indigo-500' : 'bg-white border-slate-200 text-slate-900 focus:border-indigo-500'}`}

                        disabled={!pasteText.trim()}
                        className={`w-full py-3 rounded-xl font-bold transition-all text-sm flex items-center justify-center space-x-2 ${!pasteText.trim() ? (isDarkMode ? 'bg-slate-800 text-slate-500 cursor-not-allowed' : 'bg-slate-200 text-slate-400 cursor-not-allowed') : `${theme.bg} text-white shadow-md hover:opacity-90`}`}
                      >
                        <AlignLeft size={16} />
                        <span>Parse & Add Breaks</span>
                      </button>
                    </div>
                  </div>
                )}

                {breaksTab === 'Select' && (
                  <div className={`p-5 rounded-2xl border ${isDarkMode ? 'bg-slate-950/50 border-slate-800' : 'bg-slate-50/50 border-slate-200'}`}>
                    <div className="flex items-center justify-between mb-4">
                      <h4 className={`text-[10px] font-bold uppercase tracking-widest ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Visual Selector</h4>
                      <span className={`text-[9px] font-medium ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>
                        {!visualSelection.start ? 'Click a start date' : !visualSelection.end ? 'Click an end date' : 'Range selected'}
                      </span>
                    </div>

                    {/* Mini Calendar UI */}
                    <div className="mb-4 select-none">
                      <div className="flex items-center justify-between mb-2 px-2">
                        <button 
                          onClick={() => setVisualCalDate(new Date(visualCalDate.getFullYear(), visualCalDate.getMonth() - 1, 1))}
                          className={`p-1 rounded-md transition-colors ${isDarkMode ? 'hover:bg-slate-800 text-slate-400' : 'hover:bg-slate-200 text-slate-500'}`}
                        >
                          <ChevronLeft size={16} />
                        </button>
                        <span className={`text-sm font-bold ${isDarkMode ? 'text-white' : 'text-slate-800'}`}>
                          {monthNames[visualCalDate.getMonth()]} {visualCalDate.getFullYear()}
                        </span>
                        <button 
                          onClick={() => setVisualCalDate(new Date(visualCalDate.getFullYear(), visualCalDate.getMonth() + 1, 1))}
                          className={`p-1 rounded-md transition-colors ${isDarkMode ? 'hover:bg-slate-800 text-slate-400' : 'hover:bg-slate-200 text-slate-500'}`}
                        >
                          <ChevronRight size={16} />
                        </button>
                      </div>
                      
                      <div className="grid grid-cols-7 gap-y-1 text-center">
                        {daysOfWeek.map(day => (
                          <div key={day} className={`text-[9px] font-bold uppercase tracking-wider py-1 ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>
                            {day[0]}
                          </div>
                        ))}
                        {getDaysInMonth(visualCalDate).map((cell, idx) => {
                          if (!cell) return <div key={`blank-${idx}`} className="h-8"></div>;
                          
                          const cellDateObj = new Date(visualCalDate.getFullYear(), visualCalDate.getMonth(), cell);
                          const cellTime = cellDateObj.getTime();
                          const startTime = visualSelection.start?.getTime();
                          const endTime = visualSelection.end?.getTime();
                          
                          let isStart = startTime === cellTime;
                          let isEnd = endTime === cellTime;
                          let isInRange = startTime && endTime && cellTime > startTime && cellTime < endTime;
                          let isSingleSelection = startTime === cellTime && !endTime;

                          let bgClass = "bg-transparent";
                          let textClass = isDarkMode ? "text-slate-300 hover:bg-slate-800" : "text-slate-700 hover:bg-slate-200";
                          let roundedClass = "rounded-lg";

                          if (isStart && isEnd) {
                            bgClass = theme.bg;
                            textClass = "text-white font-bold";
                          } else if (isStart) {
                            bgClass = theme.bg;
                            textClass = "text-white font-bold";
                            if (endTime) roundedClass = "rounded-l-lg rounded-r-none";
                          } else if (isEnd) {
                            bgClass = theme.bg;
                            textClass = "text-white font-bold";
                            roundedClass = "rounded-r-lg rounded-l-none";
                          } else if (isInRange) {
                            bgClass = isDarkMode ? "bg-indigo-500/20" : "bg-indigo-50";
                            textClass = isDarkMode ? "text-indigo-300 font-semibold" : "text-indigo-700 font-semibold";
                            roundedClass = "rounded-none";
                          } else if (isSingleSelection) {
                            bgClass = theme.bg;
                            textClass = "text-white font-bold";
                          }

                          return (
                            <div key={idx} className={`h-8 flex items-center justify-center`}>
                              <button
                                onClick={() => handleVisualDayClick(cellDateObj)}
                                className={`w-full h-full flex items-center justify-center text-xs transition-all ${bgClass} ${textClass} ${roundedClass}`}
                              >
                                {cell}
                              </button>
                            </div>
                          );
                        })}
                      </div>
                    </div>

                    <div className="space-y-4 border-t pt-4 mt-2 border-slate-200 dark:border-slate-800">
                      <input 
                        type="text" 
                        value={visualBreakName} 
                        onChange={(e) => setVisualBreakName(e.target.value)} 
                        placeholder="Break Label (e.g. Spring Break)" 
                        className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-900 border-slate-700 text-white focus:border-indigo-500' : 'bg-white border-slate-200 text-slate-900 focus:border-indigo-500'}`} 
                      />
                      <button 
                        onClick={handleAddVisualBreak}
                        disabled={!visualSelection.start || !visualSelection.end || !visualBreakName.trim()}
                        className={`w-full py-3 rounded-xl font-bold transition-all text-sm flex items-center justify-center space-x-2 ${(!visualSelection.start || !visualSelection.end || !visualBreakName.trim()) ? (isDarkMode ? 'bg-slate-800 text-slate-500 cursor-not-allowed' : 'bg-slate-200 text-slate-400 cursor-not-allowed') : `${theme.bg} text-white shadow-md hover:opacity-90`}`}
                      >
                        <Plus size={16} />
                        <span>Add Selected Break</span>
                      </button>
                    </div>
                  </div>
                )}

                <div className="mt-6">
                  <h4 className={`text-[10px] font-bold uppercase tracking-widest mb-3 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>
                    Active Breaks ({academicYear.breaks.length})
                  </h4>
                  {academicYear.breaks.length === 0 ? (
                    <p className={`text-xs italic ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>No breaks defined yet.</p>
                  ) : (
                    <div className="space-y-2">
                      {academicYear.breaks.map(b => (
                        <div key={b.id} className={`flex items-center justify-between p-3 rounded-xl border ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
                          <div>
                            <p className={`text-sm font-bold ${isDarkMode ? 'text-slate-200' : 'text-slate-800'}`}>{b.name}</p>
                            <p className={`text-[10px] font-bold uppercase tracking-widest mt-0.5 ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>
                              {b.startDate} to {b.endDate}
                            </p>
                          </div>
                          <button 
                            onClick={() => handleRemoveBreak(b.id)}
                            className={`p-2 rounded-lg transition-colors ${isDarkMode ? 'text-slate-500 hover:bg-rose-500/20 hover:text-rose-400' : 'text-slate-400 hover:bg-rose-50 hover:text-rose-600'}`}
                          >
                            <Trash2 size={16} />
                          </button>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              </div>
              
              <div className={`p-6 border-t ${isDarkMode ? 'border-slate-800 bg-slate-900/50' : 'border-slate-100 bg-slate-50/50'}`}>
                <button onClick={() => setShowBreaksModal(false)} className={`w-full py-3 rounded-xl font-bold transition-colors text-sm ${isDarkMode ? 'bg-slate-800 text-slate-300 hover:bg-slate-700' : 'bg-white border border-slate-200 text-slate-700 hover:bg-slate-50'}`}>
                  Done
                </button>
              </div>
            </div>
          </div>
        )}

        {/* Add/Edit Event Modal */}
        {showEventModal && (
          <div className="fixed inset-0 z-[100] flex items-center justify-center p-4">
            <div className="absolute inset-0 bg-slate-900/40 backdrop-blur-sm transition-opacity" onClick={() => setShowEventModal(false)}></div>
            <div className={`relative w-full max-w-md rounded-3xl shadow-2xl overflow-hidden flex flex-col transform transition-all scale-100 ${isDarkMode ? 'bg-slate-900 border border-slate-800' : 'bg-white border border-slate-100'}`}>
              <div className={`flex items-center justify-between p-6 border-b ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
                <h3 className={`text-lg font-bold ${isDarkMode ? 'text-white' : 'text-slate-900'}`}>{newEvent.id ? 'Edit Event' : 'Add New Event'}</h3>
                <button onClick={() => setShowEventModal(false)} className={`p-2 rounded-full transition-colors ${isDarkMode ? 'hover:bg-slate-800 text-slate-400 hover:text-white' : 'hover:bg-slate-100 text-slate-500 hover:text-slate-900'}`}>
                  <X size={18} />
                </button>
              </div>
              
              <div className="p-6 space-y-5 max-h-[70vh] overflow-y-auto">
                <div>
                  <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Event Name</label>
                  <input type="text" value={newEvent.name} onChange={(e) => setNewEvent({...newEvent, name: e.target.value})} placeholder="e.g. Staff Meeting" className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`} />
                </div>

                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Start Date</label>
                    <input 
                      type="date" 
                      value={newEvent.date} 
                      onChange={(e) => {
                        const newDate = e.target.value;
                        setNewEvent(prev => ({ 
                           ...prev, 
                           date: newDate,
                           endDate: (!prev.endDate || prev.endDate < newDate) ? newDate : prev.endDate
                        }));
                      }} 
                      className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`} 
                    />
                  </div>
                  <div>
                    <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>End Date</label>
                    <input 
                      type="date" 
                      value={newEvent.endDate} 
                      min={newEvent.date}
                      onChange={(e) => setNewEvent({...newEvent, endDate: e.target.value})} 
                      className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`} 
                    />
                  </div>
                </div>
                
                <div>
                  <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Recurring Options</label>
                  <div className="flex items-center space-x-3">
                    <button 
                      onClick={() => setNewEvent({...newEvent, isRecurring: !newEvent.isRecurring})}
                      className={`p-3 flex items-center justify-center rounded-xl border transition-all shrink-0 ${newEvent.isRecurring ? `${theme.bg} text-white border-transparent shadow-sm` : (isDarkMode ? 'bg-slate-950 border-slate-800 text-slate-400 hover:bg-slate-800 hover:text-white' : 'bg-slate-50 border-slate-200 text-slate-500 hover:bg-slate-100 hover:text-slate-900')}`}
                      title={newEvent.isRecurring ? 'Toggle Recurring Off' : 'Toggle Recurring On'}
                    >
                      <Repeat size={18} />
                    </button>
                    {newEvent.isRecurring ? (
                      <select 
                        value={newEvent.frequency} 
                        onChange={(e) => setNewEvent({...newEvent, frequency: e.target.value})} 
                        className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold cursor-pointer ${isDarkMode ? 'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`}
                      >
                        <option value="Weekly">Weekly</option>
                        <option value="Bi-Weekly">Bi-Weekly</option>
                        <option value="Monthly">Monthly</option>
                      </select>
                    ) : (
                      <div className={`w-full px-4 py-3 rounded-xl border transition-colors text-sm font-semibold flex items-center ${isDarkMode ? 'bg-slate-950/50 border-slate-800/50 text-slate-600' : 'bg-slate-50/50 border-slate-200/50 text-slate-400'}`}>
                        Does not repeat
                      </div>
                    )}
                  </div>
                </div>

                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Start Time</label>
                    <div className="flex space-x-1.5">
                      <input type="text" value={newEvent.startTime} onChange={(e) => setNewEvent({...newEvent, startTime: e.target.value})} placeholder="12:00" className={`w-full px-3 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`} />
                      <div className={`flex rounded-xl border overflow-hidden shrink-0 transition-colors ${isDarkMode ? 'border-slate-800' : 'border-slate-200'}`}>
                        <button onClick={() => setNewEvent({...newEvent, startAmpm: 'AM'})} className={`px-2.5 font-bold text-xs transition-colors ${newEvent.startAmpm === 'AM' ? theme.bg + ' text-white' : (isDarkMode ? 'bg-slate-950 text-slate-500 hover:bg-slate-800' : 'bg-slate-50 text-slate-400 hover:bg-slate-100')}`}>AM</button>
                        <button onClick={() => setNewEvent({...newEvent, startAmpm: 'PM'})} className={`px-2.5 font-bold text-xs transition-colors ${newEvent.startAmpm === 'PM' ? theme.bg + ' text-white' : (isDarkMode ? 'bg-slate-950 text-slate-500 hover:bg-slate-800' : 'bg-slate-50 text-slate-400 hover:bg-slate-100')}`}>PM</button>
                      </div>
                    </div>
                  </div>
                  <div>
                    <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>End Time</label>
                    <div className="flex space-x-1.5">
                      <input type="text" value={newEvent.endTime} onChange={(e) => setNewEvent({...newEvent, endTime: e.target.value})} placeholder="01:00" className={`w-full px-3 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`} />
                      <div className={`flex rounded-xl border overflow-hidden shrink-0 transition-colors ${isDarkMode ? 'border-slate-800' : 'border-slate-200'}`}>
                        <button onClick={() => setNewEvent({...newEvent, endAmpm: 'AM'})} className={`px-2.5 font-bold text-xs transition-colors ${newEvent.endAmpm === 'AM' ? theme.bg + ' text-white' : (isDarkMode ? 'bg-slate-950 text-slate-500 hover:bg-slate-800' : 'bg-slate-50 text-slate-400 hover:bg-slate-100')}`}>AM</button>
                        <button onClick={() => setNewEvent({...newEvent, endAmpm: 'PM'})} className={`px-2.5 font-bold text-xs transition-colors ${newEvent.endAmpm === 'PM' ? theme.bg + ' text-white' : (isDarkMode ? 'bg-slate-950 text-slate-500 hover:bg-slate-800' : 'bg-slate-50 text-slate-400 hover:bg-slate-100')}`}>PM</button>
                      </div>
                    </div>
                  </div>
                </div>
                <div>
                  <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Assign to Calendar</label>
                  <select value={newEvent.calendarId} onChange={(e) => setNewEvent({...newEvent, calendarId: e.target.value})} className={`w-full px-4 py-3 rounded-xl border outline-none appearance-none transition-colors text-sm font-semibold ${isDarkMode ? 'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`}>
                    <option value="">None (Personal)</option>
                    {customCalendars.map(cal => <option key={cal.id} value={cal.id}>{cal.name}</option>)}
                  </select>
                </div>
                <div>
                  <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Event Icon</label>
                  <div className="flex flex-wrap gap-2">
                    {Object.keys(iconOptions).map(iconKey => {
                      const IconCmp = iconOptions[iconKey];
                      const isSelected = newEvent.icon === iconKey;
                      const primaryBg = iconThemes[iconKey][0].primaryBg;
                      return (
                        <button key={iconKey} onClick={() => setNewEvent({...newEvent, icon: iconKey})} className={`p-2.5 rounded-xl border transition-all ${isSelected ? `${primaryBg} text-white border-transparent shadow-sm` : (isDarkMode ? 'border-slate-800 text-slate-400 hover:bg-slate-800 hover:text-white' : 'border-slate-200 text-slate-500 hover:bg-slate-50 hover:text-slate-900')}`}>
                          <IconCmp size={18} />
                        </button>
                      );
                    })}
                  </div>
                </div>
                <div>
                  <label className={`block text-[10px] font-bold uppercase tracking-widest mb-2 ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>Notes</label>
                  <textarea value={newEvent.notes} onChange={(e) => setNewEvent({...newEvent, notes: e.target.value})} placeholder="Add any additional details or notes here..." className={`w-full px-4 py-3 rounded-xl border outline-none transition-colors text-sm font-semibold min-h-[80px] resize-none ${isDarkMode ? 'bg-slate-950 border-slate-800 text-white focus:border-indigo-500' : 'bg-slate-50 border-slate-200 text-slate-900 focus:border-indigo-500'}`} />
                </div>
              </div>
              
              <div className={`p-6 border-t flex space-x-3 ${isDarkMode ? 'border-slate-800 bg-slate-900/50' : 'border-slate-100 bg-slate-50/50'}`}>
                <button onClick={() => setShowEventModal(false)} className={`flex-1 py-3 rounded-xl font-bold transition-colors text-sm ${isDarkMode ? 'bg-slate-800 text-slate-300 hover:bg-slate-700' : 'bg-white border border-slate-200 text-slate-700 hover:bg-slate-50'}`}>Cancel</button>
                <button onClick={handleAddEvent} className={`flex-1 py-3 rounded-xl font-bold text-white transition-colors shadow-md text-sm ${theme.bg} hover:shadow-lg`}>{newEvent.id ? 'Save Changes' : 'Add Event'}</button>
              </div>
            </div>
          </div>
        )}

        {/* Day Click Context Menu Popout */}
        {dayContextMenu && (
          <div className="fixed inset-0 z-[120]" onMouseDown={() => setDayContextMenu(null)}>
            <div className={`absolute w-48 rounded-xl shadow-[0_20px_50px_rgba(0,0,0,0.15)] border py-2 ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`} style={{ top: Math.min(dayContextMenu.y, window.innerHeight - 150), left: Math.min(dayContextMenu.x, window.innerWidth - 200) }} onMouseDown={(e) => e.stopPropagation()}>
              <div className={`px-3 py-2 mb-1 border-b text-[10px] font-bold uppercase tracking-widest ${isDarkMode ? 'border-slate-800/50 text-slate-400' : 'border-slate-50 text-slate-500'}`}>
                {monthNames[dayContextMenu.date.getMonth()]} {dayContextMenu.date.getDate()} Options
              </div>
              <button className={`w-full flex items-center px-3 py-2.5 text-xs font-semibold transition-colors ${isDarkMode ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-600 hover:bg-slate-50'}`} onClick={() => handleViewDay(dayContextMenu.date)}>
                <div className={`w-6 h-6 rounded flex items-center justify-center mr-3 ${isDarkMode ? 'bg-slate-800' : 'bg-slate-50'}`}><Calendar size={12} /></div>
                View Day
              </button>
              <button className={`w-full flex items-center px-3 py-2.5 text-xs font-semibold transition-colors ${isDarkMode ? 'text-slate-300 hover:bg-slate-800' : 'text-slate-600 hover:bg-slate-50'}`} onClick={() => { 
                const dateStr = getFormattedDate(dayContextMenu.date);
                setCalendarDate(dayContextMenu.date); 
                setNewEvent({ name: '', date: dateStr, endDate: dateStr, startTime: '12:00', startAmpm: 'PM', endTime: '01:00', endAmpm: 'PM', calendarId: '', icon: 'Calendar', notes: '', isRecurring: false, frequency: 'Weekly', excludedDates: [] }); 
                setShowEventModal(true); 
                setDayContextMenu(null); 
              }}>
                <div className={`w-6 h-6 rounded flex items-center justify-center mr-3 ${isDarkMode ? 'bg-slate-800' : 'bg-slate-50'}`}><Plus size={12} /></div>
                Add Event
              </button>
            </div>
          </div>
        )}

        {/* Event Details Modal */}
        {selectedEvent && (() => {
          const IconCmp = iconOptions[selectedEvent.icon] || Calendar;
          const cal = customCalendars.find(c => c.id === Number(selectedEvent.calendarId));
          const colorStyle = selectedEvent._colorStyle || iconThemes[selectedEvent.icon || 'Calendar'][0];

          return (
            <div className="fixed inset-0 z-[130] flex items-center justify-center p-4">
              <div className="absolute inset-0 bg-slate-900/40 backdrop-blur-sm transition-opacity" onClick={() => setSelectedEvent(null)}></div>
              <div className={`relative w-full max-w-sm rounded-3xl shadow-2xl overflow-hidden flex flex-col transform transition-all scale-100 ${isDarkMode ? 'bg-slate-900' : 'bg-white'}`}>
                <div className={`p-6 flex items-start justify-between ${colorStyle.bg} ${colorStyle.text}`}>
                  <div className="flex items-center space-x-4">
                     <div className={`${colorStyle.iconBg} p-3 rounded-2xl`}><IconCmp size={24} /></div>
                     <div>
                       <h3 className="text-xl font-bold leading-tight">{selectedEvent.name}</h3>
                       <p className="text-sm opacity-90 font-medium mt-1">
                         {selectedEvent.date === (selectedEvent.endDate || selectedEvent.date)
                           ? (selectedEvent.occurrenceDate || selectedEvent.date)
                           : `${selectedEvent.date} to ${selectedEvent.endDate || selectedEvent.date}`}
                         {selectedEvent.isRecurring && ` • ${selectedEvent.frequency}`}
                       </p>
                     </div>
                  </div>
                  <button onClick={() => setSelectedEvent(null)} className="p-2 bg-white/10 hover:bg-white/20 rounded-full transition-colors"><X size={18} /></button>
                </div>
                
                <div className="p-6 space-y-6">
                  <div className="flex items-center space-x-4">
                    <div className={`w-10 h-10 rounded-xl flex items-center justify-center ${isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-100 text-slate-500'}`}><Clock size={18} /></div>
                    <div>
                      <p className={`text-[10px] font-bold uppercase tracking-widest ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Time Segment</p>
                      <p className={`text-sm font-semibold ${isDarkMode ? 'text-slate-200' : 'text-slate-700'}`}>
                        {selectedEvent.startTime} {selectedEvent.startAmpm} - {selectedEvent.endTime} {selectedEvent.endAmpm}
                      </p>
                    </div>
                  </div>

                  {cal && (
                    <div className="flex items-center space-x-4">
                      <div className={`w-10 h-10 rounded-xl flex items-center justify-center ${isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-100 text-slate-500'}`}><Calendar size={18} /></div>
                      <div>
                        <p className={`text-[10px] font-bold uppercase tracking-widest ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Assigned Calendar</p>
                        <p className={`text-sm font-semibold ${isDarkMode ? 'text-slate-200' : 'text-slate-700'}`}>{cal.name}</p>
                      </div>
                    </div>
                  )}

                  <div className="flex items-start space-x-4">
                    <div className={`w-10 h-10 rounded-xl flex items-center justify-center shrink-0 ${isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-100 text-slate-500'}`}><AlignLeft size={18} /></div>
                    <div>
                      <p className={`text-[10px] font-bold uppercase tracking-widest mb-1 ${isDarkMode ? 'text-slate-500' : 'text-slate-400'}`}>Event Notes</p>
                      <p className={`text-sm font-medium leading-relaxed whitespace-pre-wrap ${isDarkMode ? 'text-slate-300' : 'text-slate-600'}`}>
                        {selectedEvent.notes || 'No additional notes provided.'}
                      </p>
                    </div>
                  </div>
                </div>
                
                <div className={`p-6 border-t flex space-x-3 ${isDarkMode ? 'border-slate-800 bg-slate-900/50' : 'border-slate-100 bg-slate-50/50'}`}>
                  <button onClick={() => handleInitiateDelete(selectedEvent)} className={`flex-1 py-3 rounded-xl font-bold transition-colors text-sm ${isDarkMode ? 'bg-transparent text-slate-400 hover:bg-slate-800 hover:text-slate-200' : 'bg-transparent text-slate-500 hover:bg-slate-100 hover:text-slate-800'}`}>
                    Cancel Event
                  </button>
                  <button onClick={() => handleEditEvent(selectedEvent)} className={`flex-1 py-3 rounded-xl font-bold text-white transition-colors shadow-md text-sm ${theme.bg} hover:shadow-lg`}>
                    Edit Event
                  </button>
                </div>
              </div>
            </div>
          );
        })()}

      </div>
    </div>
  );
};

export default App;
