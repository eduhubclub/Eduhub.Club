const BubbleView = ({ theme, isDarkMode, hasPermission, startMonitoring, dbLevel, currentDbRef, addBubbleTrigger, isFullScreen, setIsFullScreen, sensitivity, setSensitivity }) => {
  const containerRef = useRef(null);
  const canvasRef = useRef(null);
  const bubblesRef = useRef([]);
  const lastPopTimeRef = useRef(0);

  // Handle Resizing
  useEffect(() => {
    if (!containerRef.current || !canvasRef.current) return;
    
    // ResizeObserver watches the actual element dimensions frame-by-frame,
    // which perfectly handles the CSS transition stretching for full-screen!
    const resizeObserver = new ResizeObserver((entries) => {
      for (let entry of entries) {
        if (canvasRef.current) {
            canvasRef.current.width = entry.contentRect.width;
            canvasRef.current.height = entry.contentRect.height;
        }
      }
    });

    resizeObserver.observe(containerRef.current);

    return () => resizeObserver.disconnect();
  }, [hasPermission, isFullScreen]);

  // Handle Adding Bubbles manually
  useEffect(() => {
    if (addBubbleTrigger > 0 && canvasRef.current) {
      const radius = Math.random() * 20 + 20; // 20 to 40px
      const canvas = canvasRef.current;
      bubblesRef.current.push({
        x: Math.random() * (canvas.width - radius * 2) + radius,
        y: canvas.height - radius - 10,
        vx: (Math.random() - 0.5) * 3,
        vy: (Math.random() * -2) - 1, // Move upwards
        radius: radius,
        popping: false,
        popProgress: 0,
        hue: Math.floor(Math.random() * 360)
      });
    }
  }, [addBubbleTrigger]);

  // Bubble Animation Loop
  useEffect(() => {
    if (!hasPermission) return;
    let animationId;
    
    const render = () => {
      const canvas = canvasRef.current;
      if (!canvas) return;
      const ctx = canvas.getContext('2d');
      
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      
      // Read directly from the highly responsive ref instead of throttled state
      const currentDb = currentDbRef.current;
      const now = Date.now();

      // Pop logic if too loud
      if (currentDb >= 65) {
        // Pop exactly one bubble, with a 2-second cooldown so they don't all pop instantly
        if (now - lastPopTimeRef.current > 2000) {
          const unpopped = bubblesRef.current.filter(b => !b.popping);
          if (unpopped.length > 0) {
            const randomBubble = unpopped[Math.floor(Math.random() * unpopped.length)];
            randomBubble.popping = true;
            lastPopTimeRef.current = now;
          }
        }
      }

      for (let i = bubblesRef.current.length - 1; i >= 0; i--) {
        const b = bubblesRef.current[i];

        if (b.popping) {
           b.popProgress += 0.08;
           b.radius += 1.5;
           
           ctx.beginPath();
           ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2);
           ctx.strokeStyle = `rgba(255, 255, 255, ${1 - b.popProgress})`;
           ctx.lineWidth = 2;
           ctx.stroke();

           // Splash particles
           for(let p=0; p<4; p++) {
             ctx.fillStyle = `rgba(255, 255, 255, ${(1 - b.popProgress) * 0.5})`;
             ctx.beginPath();
             ctx.arc(b.x + Math.cos(p * Math.PI/2) * b.radius * 0.8, b.y + Math.sin(p * Math.PI/2) * b.radius * 0.8, 3, 0, Math.PI*2);
             ctx.fill();
           }

           if (b.popProgress >= 1) {
             bubblesRef.current.splice(i, 1);
           }
        } else {
           // Float Physics
           b.vy -= 0.01; // Buoyancy (gravity goes up)
           b.vx += (Math.random() - 0.5) * 0.1; // Drift
           
           // Speed limits
           if (b.vx > 2) b.vx = 2;
           if (b.vx < -2) b.vx = -2;
           if (b.vy < -3) b.vy = -3;
           if (b.vy > 2) b.vy = 2;

           b.x += b.vx;
           b.y += b.vy;

           // Bounce off walls
           if (b.x + b.radius > canvas.width) { b.x = canvas.width - b.radius; b.vx *= -0.8; }
           if (b.x - b.radius < 0) { b.x = b.radius; b.vx *= -0.8; }
           if (b.y + b.radius > canvas.height) { b.y = canvas.height - b.radius; b.vy *= -0.8; }
           if (b.y - b.radius < 0) { b.y = b.radius; b.vy *= -0.8; }

           // Draw bubble
           ctx.beginPath();
           ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2);
           ctx.fillStyle = `hsla(${b.hue}, 80%, 60%, 0.15)`;
           ctx.fill();
           
           // Border
           ctx.strokeStyle = `hsla(${b.hue}, 80%, 70%, 0.6)`;
           ctx.lineWidth = 1.5;
           ctx.stroke();

           // White Reflection Highlight
           ctx.beginPath();
           ctx.arc(b.x - b.radius*0.3, b.y - b.radius*0.3, b.radius*0.2, 0, Math.PI * 2);
           ctx.fillStyle = 'rgba(255,255,255,0.4)';
           ctx.fill();
        }
      }

      animationId = requestAnimationFrame(render);
    };

    render();
    return () => cancelAnimationFrame(animationId);
  }, [hasPermission]);

  return (
    <div className={`${isFullScreen ? 'fixed inset-0 z-[9999] rounded-none border-0' : 'relative w-full max-w-5xl mx-auto rounded-2xl border min-h-[600px]'} flex flex-col overflow-hidden transition-all duration-300 ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
      <button
        onClick={() => setIsFullScreen(!isFullScreen)}
        className={`absolute top-4 right-4 z-50 p-2 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:bg-slate-800 hover:text-white' : 'text-slate-500 hover:bg-slate-100 hover:text-slate-900'}`}
        title={isFullScreen ? "Exit Full Screen" : "Full Screen"}
      >
        {isFullScreen ? <Minimize size={20} /> : <Maximize size={20} />}
      </button>

      {!hasPermission ? (
         <div className="p-8 h-full flex flex-col">
            <MicPrompt isDarkMode={isDarkMode} theme={theme} onStart={startMonitoring} />
         </div>
      ) : (
         <>
          <div className="absolute top-6 left-0 right-0 z-10 flex flex-col items-center pointer-events-none">
            <h2 className={`text-2xl font-bold tracking-tight mb-1 ${dbLevel >= 65 ? 'text-red-500 animate-pulse' : (isDarkMode ? 'text-white' : 'text-slate-800')}`}>
              Bubble Classroom
            </h2>
            <p className={`text-sm font-medium ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>
              Keep voices low to keep the bubbles floating!
            </p>
          </div>
          <div ref={containerRef} className="flex-1 w-full relative bg-gradient-to-b from-transparent to-slate-50/50 dark:to-slate-900/50">
            <canvas ref={canvasRef} className="absolute inset-0 w-full h-full block" />
          </div>

          {/* Sensitivity Slider for Bubble View */}
          <div className={`p-6 pr-24 md:pr-32 border-t shrink-0 z-20 ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
            <div className="flex items-center justify-between mb-2">
              <label className={`text-sm font-bold ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Microphone Sensitivity</label>
              <span className={`text-xs font-medium px-2 py-1 rounded-md ${isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-100 text-slate-500'}`}>{Math.round(sensitivity)}%</span>
            </div>
            <input type="range" min="10" max="300" value={sensitivity} onChange={(e) => setSensitivity(parseInt(e.target.value))} className={`w-full h-2 rounded-lg appearance-none cursor-pointer ${isDarkMode ? 'bg-slate-800' : 'bg-slate-200'}`} style={{ accentColor: '#f59e0b' }} />
          </div>
         </>
      )}
    </div>
  );
};

const TowerView = ({ theme, isDarkMode, hasPermission, startMonitoring, dbLevel, currentDbRef, rebuildTowerTrigger, isFullScreen, setIsFullScreen, sensitivity, setSensitivity }) => {
  const containerRef = useRef(null);
  const canvasRef = useRef(null);
  const blocksRef = useRef([]);
  const windRef = useRef([]);
  const isCollapsedRef = useRef(false);

  const initTower = () => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const groundY = canvas.height - 60;
    const numBlocks = 12;
    const blockW = Math.min(80, canvas.width / 5);
    const blockH = 40;
    
    isCollapsedRef.current = false;
    blocksRef.current = Array.from({length: numBlocks}).map((_, i) => {
      // i = 0 is bottom, i = 11 is top
      const hue = 210 - (i * 15); // Gradient from Blue to Pink
      return {
        id: i,
        baseX: canvas.width / 2,
        baseY: groundY - (i * blockH) - (blockH / 2),
        x: canvas.width / 2,
        y: groundY - (i * blockH) - (blockH / 2),
        width: blockW,
        height: blockH,
        vx: 0, vy: 0, va: 0, angle: 0,
        color: `hsl(${hue}, 80%, 55%)`,
        borderColor: `hsl(${hue}, 80%, 40%)`
      };
    });

    // Init wind particles
    windRef.current = Array.from({length: 40}).map(() => ({
      x: Math.random() * canvas.width,
      y: Math.random() * canvas.height,
      length: Math.random() * 60 + 20,
      speed: Math.random() * 8 + 4,
      thickness: Math.random() * 2 + 1
    }));
  };

  useEffect(() => {
    initTower();
  }, [rebuildTowerTrigger]);

  useEffect(() => {
    if (!containerRef.current || !canvasRef.current) return;
    
    const resizeObserver = new ResizeObserver((entries) => {
      for (let entry of entries) {
        if (canvasRef.current) {
            canvasRef.current.width = entry.contentRect.width;
            canvasRef.current.height = entry.contentRect.height;
            // Neatly restack if not currently tumbling
            if (!isCollapsedRef.current) {
                initTower(); 
            }
        }
      }
    });

    resizeObserver.observe(containerRef.current);
    return () => resizeObserver.disconnect();
  }, [hasPermission, isFullScreen]);

  // Animation and Physics loop
  useEffect(() => {
    if (!hasPermission) return;
    let animationId;
    
    const render = () => {
      const canvas = canvasRef.current;
      if (!canvas) return;
      const ctx = canvas.getContext('2d');
      const currentDb = currentDbRef.current;
      const groundY = canvas.height - 60;
      
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      
      // Draw Ground
      ctx.fillStyle = isDarkMode ? '#1e293b' : '#f1f5f9';
      ctx.fillRect(0, groundY, canvas.width, canvas.height - groundY);
      ctx.strokeStyle = isDarkMode ? '#334155' : '#e2e8f0';
      ctx.lineWidth = 4;
      ctx.beginPath();
      ctx.moveTo(0, groundY);
      ctx.lineTo(canvas.width, groundY);
      ctx.stroke();

      // Wind physics and rendering
      const windStress = Math.max(0.1, currentDb / 65);
      const isBlowingHard = currentDb >= 55;
      
      ctx.strokeStyle = isDarkMode 
        ? `rgba(255, 255, 255, ${isBlowingHard ? 0.3 : 0.1})` 
        : `rgba(0, 0, 0, ${isBlowingHard ? 0.15 : 0.05})`;
        
      windRef.current.forEach(p => {
         p.x += p.speed * (windStress * 3);
         if (p.x > canvas.width) {
             p.x = -p.length;
             p.y = Math.random() * groundY;
         }
         ctx.lineWidth = p.thickness;
         ctx.beginPath();
         ctx.moveTo(p.x, p.y);
         ctx.lineTo(p.x + p.length, p.y);
         ctx.stroke();
      });

      // Blow tower over if too loud!
      if (currentDb >= 65 && !isCollapsedRef.current) {
         isCollapsedRef.current = true;
         blocksRef.current.forEach((b, i) => {
             // Top blocks get hit harder by the wind
             const heightFactor = (i + 1) / blocksRef.current.length;
             b.vx = (8 + Math.random() * 12) * heightFactor; // Blow to the right
             b.vy = -2 - Math.random() * 4 * heightFactor; // Slight lift
             b.va = (Math.random() - 0.5) * 0.5; // Tumble
         });
      }

      // Draw blocks
      blocksRef.current.forEach((b, i) => {
         if (!isCollapsedRef.current) {
             // Wobble aggressively as it gets louder
             const stressFactor = Math.pow(currentDb / 65, 3); // Exponential wobble curve
             const heightFactor = i / blocksRef.current.length;
             const jitterX = (Math.random() - 0.5) * 15 * stressFactor * heightFactor;
             const jitterAngle = (Math.random() - 0.5) * 0.2 * stressFactor * heightFactor;
             
             b.x = b.baseX + jitterX;
             b.angle = jitterAngle;
         } else {
             // Falling physics
             b.vy += 0.5; // Gravity
             b.x += b.vx;
             b.y += b.vy;
             b.angle += b.va;

             // Ground Collision
             if (b.y + b.height/2 > groundY) {
                 b.y = groundY - b.height/2;
                 b.vy *= -0.3; // Dampen bounce
                 b.vx *= 0.8;  // Friction
                 b.va *= 0.8;  // Reduce tumbling
             }
             // Wall collisions
             if (b.x + b.width/2 > canvas.width) {
                 b.x = canvas.width - b.width/2;
                 b.vx *= -0.5;
             }
             if (b.x - b.width/2 < 0) {
                 b.x = b.width/2;
                 b.vx *= -0.5;
             }
         }

         ctx.save();
         ctx.translate(b.x, b.y);
         ctx.rotate(b.angle);
         ctx.fillStyle = b.color;
         ctx.fillRect(-b.width/2, -b.height/2, b.width, b.height);
         ctx.strokeStyle = b.borderColor;
         ctx.lineWidth = 2;
         ctx.strokeRect(-b.width/2, -b.height/2, b.width, b.height);
         
         // Inner highlight for a blocky 3D effect
         ctx.fillStyle = 'rgba(255,255,255,0.15)';
         ctx.fillRect(-b.width/2 + 2, -b.height/2 + 2, b.width - 4, 6);
         
         ctx.restore();
      });

      animationId = requestAnimationFrame(render);
    };

    render();
    return () => cancelAnimationFrame(animationId);
  }, [hasPermission]);

  return (
    <div className={`${isFullScreen ? 'fixed inset-0 z-[9999] rounded-none border-0' : 'relative w-full max-w-5xl mx-auto rounded-2xl border min-h-[600px]'} flex flex-col overflow-hidden transition-all duration-300 ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
      <button
        onClick={() => setIsFullScreen(!isFullScreen)}
        className={`absolute top-4 right-4 z-50 p-2 rounded-full transition-colors ${isDarkMode ? 'text-slate-400 hover:bg-slate-800 hover:text-white' : 'text-slate-500 hover:bg-slate-100 hover:text-slate-900'}`}
        title={isFullScreen ? "Exit Full Screen" : "Full Screen"}
      >
        {isFullScreen ? <Minimize size={20} /> : <Maximize size={20} />}
      </button>

      {!hasPermission ? (
         <div className="p-8 h-full flex flex-col">
            <MicPrompt isDarkMode={isDarkMode} theme={theme} onStart={startMonitoring} />
         </div>
      ) : (
         <>
          <div className="absolute top-6 left-0 right-0 z-10 flex flex-col items-center pointer-events-none">
            <h2 className={`text-2xl font-bold tracking-tight mb-1 ${dbLevel >= 65 ? 'text-red-500 animate-pulse' : (isDarkMode ? 'text-white' : 'text-slate-800')}`}>
              Block Tower
            </h2>
            <p className={`text-sm font-medium ${isDarkMode ? 'text-slate-400' : 'text-slate-500'}`}>
              Keep voices low so the tower doesn't blow over!
            </p>
          </div>
          <div ref={containerRef} className="flex-1 w-full relative bg-gradient-to-b from-transparent to-slate-50/50 dark:to-slate-900/50">
            <canvas ref={canvasRef} className="absolute inset-0 w-full h-full block" />
          </div>

          <div className={`p-6 pr-24 md:pr-32 border-t shrink-0 z-20 ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
            <div className="flex items-center justify-between mb-2">
              <label className={`text-sm font-bold ${isDarkMode ? 'text-slate-300' : 'text-slate-700'}`}>Microphone Sensitivity</label>
              <span className={`text-xs font-medium px-2 py-1 rounded-md ${isDarkMode ? 'bg-slate-800 text-slate-400' : 'bg-slate-100 text-slate-500'}`}>{Math.round(sensitivity)}%</span>
            </div>
            <input type="range" min="10" max="300" value={sensitivity} onChange={(e) => setSensitivity(parseInt(e.target.value))} className={`w-full h-2 rounded-lg appearance-none cursor-pointer ${isDarkMode ? 'bg-slate-800' : 'bg-slate-200'}`} style={{ accentColor: '#f59e0b' }} />
          </div>
         </>
      )}
    </div>
  );
};

const App = () => {
  const [isSidebarOpen, setIsSidebarOpen] = useState(true);
  const [sidebarSide, setSidebarSide] = useState('left');
  const [isDarkMode, setIsDarkMode] = useState(false);
  const [activeTab, setActiveTab] = useState('Tower'); // Defaulted to Tower for preview
  const [expandedItem, setExpandedItem] = useState(null); 
  const [popoutItem, setPopoutItem] = useState(null); 
  const [panelItem, setPanelItem] = useState(null); 
  const [isSearchExpanded, setIsSearchExpanded] = useState(false);
  const [searchQuery, setSearchQuery] = useState("");
  const [useOverlaySearch, setUseOverlaySearch] = useState(false);
  
  // Audio Global State
  const [hasPermission, setHasPermission] = useState(false);
  const [dbLevel, setDbLevel] = useState(0);
  const [sensitivity, setSensitivity] = useState(100); 
  const [isPaused, setIsPaused] = useState(false);
  const [addBubbleTrigger, setAddBubbleTrigger] = useState(0);
  const [rebuildTowerTrigger, setRebuildTowerTrigger] = useState(0);
  const [isFullScreen, setIsFullScreen] = useState(false);

  const audioContextRef = useRef(null);
  const analyserRef = useRef(null);
  const streamRef = useRef(null);
  const animationRef = useRef(null);
  const lastDingTime = useRef(0);
  const lastUiUpdateRef = useRef(0);
  const currentDbRef = useRef(0);
  const sensitivityRef = useRef(sensitivity);
  const isPausedRef = useRef(isPaused);

  const currentAppName = "NoiseMeter"; 
  const appThemes = {
    "NoiseMeter": { primary: "amber-500", bg: "bg-amber-500", text: "text-amber-500", border: "border-amber-500", hoverBg: "hover:bg-amber-50", ring: "ring-amber-500" }
  };
  const theme = appThemes[currentAppName];
  const popoutRef = useRef(null);
  const searchRef = useRef(null);
  const isLeft = sidebarSide === 'left';

  useEffect(() => {
    sensitivityRef.current = sensitivity;
  }, [sensitivity]);

  useEffect(() => {
    isPausedRef.current = isPaused;
  }, [isPaused]);

  useEffect(() => {
    const handleResize = () => setUseOverlaySearch(window.innerWidth < 768);
    handleResize();
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, []);

  useEffect(() => {
    const handleClickOutside = (event) => {
      if (popoutRef.current && !popoutRef.current.contains(event.target)) setPopoutItem(null);
      if (searchRef.current && !searchRef.current.contains(event.target) && !searchQuery) setIsSearchExpanded(false);
    };
    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, [searchQuery]);

  const playPop = () => {
    if (!audioContextRef.current) return;
    const ctx = audioContextRef.current;
    const osc = ctx.createOscillator();
    const gainNode = ctx.createGain();
    
    osc.type = 'sine';
    // Start high and rapidly drop pitch to simulate a "pop" / "bloop"
    osc.frequency.setValueAtTime(800, ctx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(100, ctx.currentTime + 0.08);
    
    // Very quick attack and decay for the percussive pop sound
    gainNode.gain.setValueAtTime(0, ctx.currentTime);
    gainNode.gain.linearRampToValueAtTime(1, ctx.currentTime + 0.01);
    gainNode.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.1);
    
    osc.connect(gainNode);
    gainNode.connect(ctx.destination);
    
    osc.start();
    osc.stop(ctx.currentTime + 0.1);
  };

  const startMonitoring = async () => {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
      streamRef.current = stream;
      setHasPermission(true);

      const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      audioContextRef.current = audioCtx;
      
      const analyser = audioCtx.createAnalyser();
      analyser.fftSize = 512;
      analyserRef.current = analyser;

      const source = audioCtx.createMediaStreamSource(stream);
      source.connect(analyser);

      const dataArray = new Uint8Array(analyser.fftSize);

      const updateMeter = () => {
        if (!isPausedRef.current) {
          analyser.getByteTimeDomainData(dataArray);
          let sumSquares = 0;
          for(let i = 0; i < dataArray.length; i++) {
              let norm = (dataArray[i] / 128.0) - 1.0;
              sumSquares += norm * norm;
          }
          let rms = Math.sqrt(sumSquares / dataArray.length);
          let volume = rms * 350 * (sensitivityRef.current / 100);
          let finalDb = Math.min(100, Math.max(0, volume));
          
          currentDbRef.current = finalDb;

          const now = Date.now();
          // Throttle React state updates to prevent UI rendering lag
          if (now - lastUiUpdateRef.current > 50) {
              setDbLevel(finalDb);
              lastUiUpdateRef.current = now;
          }

          // Alert Pop Threshold
          if (finalDb >= 65) {
              // Synced to 2 seconds to perfectly match the visual bubble pop cooldown
              if (now - lastDingTime.current > 2000) {
                  playPop();
                  lastDingTime.current = now;
              }
          }
        }
        animationRef.current = requestAnimationFrame(updateMeter);
      };
      updateMeter();
    } catch (err) {
      setHasPermission(false);
    }
  };

  // Cleanup Global Audio
  useEffect(() => {
    return () => {
        if (animationRef.current) cancelAnimationFrame(animationRef.current);
        if (audioContextRef.current && audioContextRef.current.state !== 'closed') {
            audioContextRef.current.close();
        }
        if (streamRef.current) {
            streamRef.current.getTracks().forEach(track => track.stop());
        }
    };
  }, []);

  const navigationItems = [
    { id: 'noisemeter', name: 'Noise Meter', icon: Mic, type: 'link' },
    { id: 'bubble', name: 'Bubble', icon: Circle, type: 'link' },
    { id: 'tower', name: 'Tower', icon: Building, type: 'link' }
  ];

  const handleItemClick = (item) => {
    if (item.type === 'link') {
      setActiveTab(item.name);
      setExpandedItem(null);
      setPanelItem(null);
    }
  };

  return (
    <div className={`flex min-h-screen font-sans transition-all duration-300 ${isDarkMode ? 'bg-slate-950 text-slate-100' : 'bg-slate-50 text-slate-900'} ${isLeft ? 'flex-row' : 'flex-row-reverse'}`}>
      
      {/* SIDEBAR */}
      <aside className={`relative transition-all duration-300 ease-in-out flex flex-col shadow-sm ${isSidebarOpen ? 'w-64' : 'w-20'} ${isLeft ? 'border-r' : 'border-l'} ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'} ${isFullScreen ? 'z-0' : 'z-50'}`}>
        <div className={`flex items-center justify-center shrink-0 border-b h-16 w-full ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
          <div className="flex items-center justify-center px-4">
            {isSidebarOpen ? <LogoHorizontal className="h-6 w-auto" /> : <LogoIcon2x2 className="w-8 h-8" />}
          </div>
        </div>
        <nav className="flex-1 px-2 space-y-1 pt-2 pb-10 relative overflow-x-visible">
          {navigationItems.map((item) => (
            <div key={item.id} className="relative">
              <button onClick={() => handleItemClick(item)} className={`flex items-center transition-all group relative rounded-xl h-12 ${isSidebarOpen ? 'w-full px-3' : 'w-16 mx-auto justify-center'} ${activeTab === item.name ? `${theme.bg} text-white shadow-lg` : `${isDarkMode ? 'text-slate-400 hover:bg-slate-800 hover:text-slate-200' : 'text-slate-500 hover:bg-slate-100 hover:text-slate-900'}`}`}>
                <div className="flex items-center justify-center shrink-0"><item.icon size={20} strokeWidth={activeTab === item.name ? 2.5 : 2} /></div>
                {isSidebarOpen && <span className="ml-4 font-semibold text-sm flex-1 text-left truncate">{item.name}</span>}
              </button>
            </div>
          ))}
        </nav>
        <div className={`p-2 border-t ${isDarkMode ? 'border-slate-800' : 'border-slate-100'}`}>
           <div className={`flex transition-all duration-300 ${isSidebarOpen ? 'flex-row items-center space-x-2' : 'flex-col items-center space-y-2'}`}>
              <button onClick={() => setIsDarkMode(!isDarkMode)} className={`flex items-center justify-center rounded-xl transition-all h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'w-16'} ${isDarkMode ? 'text-amber-400 bg-slate-800 hover:bg-slate-700' : 'text-slate-400 bg-slate-50 hover:bg-slate-100 hover:text-slate-900'}`}>
                {isDarkMode ? <Sun size={18} /> : <Moon size={18} />}
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Dark</span>}
              </button>
              <button onClick={() => setSidebarSide(isLeft ? 'right' : 'left')} className={`flex items-center justify-center rounded-xl transition-all h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'w-16'} ${isDarkMode ? 'text-slate-500 hover:bg-slate-800 hover:text-slate-200' : 'text-slate-400 hover:text-slate-900 hover:bg-slate-100'}`}>
                <ArrowLeftRight size={16} />
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Flip</span>}
              </button>
              <button onClick={() => setIsSidebarOpen(!isSidebarOpen)} className={`flex items-center justify-center transition-all rounded-xl h-12 ${isSidebarOpen ? 'flex-1 px-3' : 'mx-auto w-16'} ${isDarkMode ? 'text-slate-400 bg-slate-800/40 hover:bg-slate-700/60 hover:text-slate-200' : 'text-slate-400 bg-slate-50 hover:bg-slate-100 hover:text-slate-900'}`}>
                {isSidebarOpen ? <ChevronLeft size={18} className={!isLeft ? 'rotate-180' : ''} /> : <Menu size={20} />}
                {isSidebarOpen && <span className="ml-2 text-[9px] font-black uppercase tracking-tighter whitespace-nowrap">Menu</span>}
              </button>
           </div>
        </div>
      </aside>

      {/* MAIN LAYOUT WRAPPER */}
      <div className={`flex-1 flex flex-col min-w-0 relative ${isFullScreen ? 'z-[100]' : 'z-10'}`}>
        <header className={`border-b flex items-center px-6 z-20 shrink-0 h-16 sticky top-0 transition-colors duration-300 ${isDarkMode ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'}`}>
          <div className="flex items-center text-lg font-bold whitespace-nowrap tracking-tight">
            <span className={isDarkMode ? 'text-white' : 'text-slate-900'}>Edu.</span>
            <span className={theme.text}>NoiseMeter</span>
          </div>
          <div className="flex items-center justify-end flex-1 h-full relative">
             <div className="flex items-center space-x-2 shrink-0">
                <button className={`p-2 text-slate-400 hover:${theme.text} rounded-full transition-colors`}><Grid size={18} /></button>
                <button className={`p-2 text-slate-400 hover:${theme.text} rounded-full transition-colors relative`}><Bell size={18} /><span className={`absolute top-2.5 right-2.5 w-1.5 h-1.5 ${theme.bg} rounded-full border border-white`}></span></button>
                <div className={`h-6 w-[1px] mx-1 shrink-0 ${isDarkMode ? 'bg-slate-800' : 'bg-slate-200'}`}></div>
                <button className={`w-9 h-9 rounded-full flex items-center justify-center transition-all shadow-sm ${isDarkMode ? 'bg-slate-800 border-slate-700 text-slate-400 hover:text-white hover:border-slate-600' : 'bg-slate-50 border-slate-200 text-slate-400 hover:bg-white hover:shadow-md'}`}><User size={20} strokeWidth={2} /></button>
             </div>
          </div>
        </header>

        {/* SCROLLABLE MAIN CONTENT */}
        <main className="flex-1 overflow-auto p-6 md:p-10 relative flex flex-col">
          {activeTab === 'Noise Meter' && (
             <NoiseMeterCard 
                theme={theme} 
                isDarkMode={isDarkMode} 
                hasPermission={hasPermission}
                dbLevel={dbLevel}
                sensitivity={sensitivity}
                setSensitivity={setSensitivity}
                isPaused={isPaused}
                setIsPaused={setIsPaused}
                startMonitoring={startMonitoring}
                isFullScreen={isFullScreen}
                setIsFullScreen={setIsFullScreen}
             />
          )}

          {activeTab === 'Bubble' && (
             <BubbleView 
                theme={theme} 
                isDarkMode={isDarkMode} 
                hasPermission={hasPermission}
                startMonitoring={startMonitoring}
                dbLevel={dbLevel}
                currentDbRef={currentDbRef}
                addBubbleTrigger={addBubbleTrigger}
                isFullScreen={isFullScreen}
                setIsFullScreen={setIsFullScreen}
                sensitivity={sensitivity}
                setSensitivity={setSensitivity}
             />
          )}

          {activeTab === 'Tower' && (
             <TowerView 
                theme={theme} 
                isDarkMode={isDarkMode} 
                hasPermission={hasPermission}
                startMonitoring={startMonitoring}
                dbLevel={dbLevel}
                currentDbRef={currentDbRef}
                rebuildTowerTrigger={rebuildTowerTrigger}
                isFullScreen={isFullScreen}
                setIsFullScreen={setIsFullScreen}
                sensitivity={sensitivity}
                setSensitivity={setSensitivity}
             />
          )}
          
          {/* Fallback empty state */}
          {activeTab !== 'Noise Meter' && activeTab !== 'Bubble' && activeTab !== 'Tower' && (
            <div className="flex items-center justify-center h-full text-slate-400">
              Select an item from the sidebar
            </div>
          )}
        </main>

        {/* FIXED FLOATING ACTION BUTTON - ONLY visible when Bubble tab is active */}
        {activeTab === 'Bubble' && hasPermission && (
          <button 
            onClick={() => setAddBubbleTrigger(prev => prev + 1)}
            className={`fixed bottom-8 right-8 md:bottom-12 md:right-12 w-14 h-14 rounded-full shadow-[0_10px_25px_-5px_rgba(0,0,0,0.3)] hover:shadow-xl flex items-center justify-center text-white transition-all transform hover:scale-105 z-[10000] ${theme.bg} hover:brightness-110 active:scale-95`}
            title="Add Bubble"
          >
            <Plus size={28} strokeWidth={2.5} />
          </button>
        )}

        {/* FIXED FLOATING ACTION BUTTON - ONLY visible when Tower tab is active */}
        {activeTab === 'Tower' && hasPermission && (
          <button 
            onClick={() => setRebuildTowerTrigger(prev => prev + 1)}
            className={`fixed bottom-8 right-8 md:bottom-12 md:right-12 w-14 h-14 rounded-full shadow-[0_10px_25px_-5px_rgba(0,0,0,0.3)] hover:shadow-xl flex items-center justify-center text-white transition-all transform hover:scale-105 z-[10000] ${theme.bg} hover:brightness-110 active:scale-95`}
            title="Rebuild Tower"
          >
            <RefreshCw size={28} strokeWidth={2.5} />
          </button>
        )}
      </div>
    </div>
  );
};

export default App;
